    60→/**

    62→ * projection of the IndexedDB event log), the theme engine, screen switching
    63→ * and all mutations via `dispatch` — every change is an appended event (ADR
    64→ * 0002). Persistence: event log -> projection -> snapshot mirror.
    65→ */
    66→export function LumioApp() {
    67→  const [state, setState] = useState<LumioState | null>(null);
    68→  const [hydrated, setHydrated] = useState(false);

    81→  const [studyBuddyOpen, setStudyBuddyOpen] = useState(false);

    85→  useEffect(() => { stateRef.current = state; }, [state]);
    86→

    88→  // Plain constants (no useMemo). Keeping every binding above the effects
    89→  // guarantees no hoisted effect can ever read them in their temporal dead zone
    90→  // (the chunk-1 TDZ bug — this ordering is a hard rule for this file).
    91→  const tz = state?.profile?.tz ?? DEFAULT_TZ;
    92→  const calm = state?.settings.calm ?? false;

    94→  const themeSchedule = state?.settings.sch ?? DEFAULT_SETTINGS.sch;

    96→    themeOverride !== 'auto' ? themeOverride : themeAt(hourOf(now, tz), themeSchedule);
    97→  const band = state?.profile?.band;
    98→  const density =

   101→  // During first-run onboarding there is no profile yet, so the wizard draft's
   102→  // language answer applies live (question 2 of onboarding).
   103→  const draftLang = (state?.draft as unknown as WizardDraft | null)?.lang;

   106→  // chunk 5B: AI features are hidden entirely until the capability probe says otherwise
   107→  const aiAvailable = aiMode === 'fake' || aiMode === 'zai' || aiMode === 'openai-compatible';

   109→  // hydrate once (client only; never during SSR): replay the event log, seeded
   110→  // from the chunk-1 snapshot when present. Falls back to the snapshot mirror
   111→  // if IndexedDB replay fails entirely.
   112→  useEffect(() => {
   113→    let alive = true;
   114→    requestPersistence();
   115→    queueMicrotask(async () => {
   116→      if (!alive) return;
   117→      try {
   118→        const events = await loadEventsAndSeed();
   119→        const projected = replayEvents(events, new Date(), events.length && (events[events.length - 1] as { tz?: string }).tz ? ((events[events.length - 1] as unknown as { tz: string }).tz) : DEFAULT_TZ);
   120→        if (alive) {
   121→          setState(projected);
   122→          saveState(projected);
   123→        }
   124→      } catch {
   125→        const snapshot = loadState();
   126→        if (alive) setState(snapshot ? projectState(normalizeState(snapshot), new Date(), snapshot.profile?.tz ?? DEFAULT_TZ) : defaultState());
   127→      }
   128→      if (alive) setHydrated(true);
   129→    });
   130→    return () => { alive = false; };
   131→  }, []);
   132→
   133→  // minute ticker: keeps the theme engine and scheduler honest without tick-count timers
   134→  useEffect(() => {
   135→    const t = setInterval(() => setNow(new Date()), 30_000);

   137→  }, []);
   138→
   139→  // Theme + density classes must live on <html> so that font-size (rem scale),

   143→    const root = document.documentElement;
   144→    root.classList.remove(
   145→      'theme-morning', 'theme-evening', 'theme-night', 'night',

   148→    root.classList.add(`theme-${theme}`, `density-${density}`);
   149→    if (theme === 'night') root.classList.add('night');

   152→
   153→  // Language class on <html> pairs Noto Sans Devanagari with the brand fonts
   154→  // and keeps assistive tech informed (chunk 4B i18n).

   156→    const root = document.documentElement;
   157→    root.classList.remove('lang-hi', 'lang-hinglish', 'lang-en');

   161→
   162→  // ---- chunk 5A: account probe — AFTER hydration, never blocking ----

   165→  // every existing feature keeps working (degradation matrix master 8.2).
   166→  useEffect(() => {

   170→      const caps = await authApi.capabilities();

   173→      const unavailable = !caps.ok || !me.ok || me.data.unavailable === true;

   176→        setDevOutbox(caps.data.devOutbox === true);
   177→        const mode = caps.data.ai?.mode;
   178→        setAiMode(mode === 'fake' || mode === 'zai' || mode === 'openai-compatible' ? mode : 'off');
   179→      } else {
   180→        setAiMode('off');
   181→      }
   182→      if (me.ok && me.data.signedIn && me.data.user) {
   183→        setAccountUser(me.data.user);
   184→        await dbMetaSet('account-user-id', me.data.user.id);
   185→      } else {
   186→        setAccountUser(null);
   187→      }
   188→    })();
   189→    return () => { alive = false; };
   190→  }, [hydrated]);
   191→
   192→  const recordAccountId = (id: string) => {
   193→    try {
   194→      localStorage.setItem('lumio-account-last', id);
   195→    } catch {
   196→      /* private mode */
   197→    }
   198→  };
   199→
   200→  const handleSignedIn = (user: AccountUser) => {
   201→    const prevId = (() => {
   202→      try {
   203→        return localStorage.getItem('lumio-account-last');
   204→      } catch {
   205→        return null;
   206→      }
   207→    })();
   208→    const hasLocalProgress = !!stateRef.current?.profile;
   209→    if (prevId && prevId !== user.id && hasLocalProgress) {
   210→      // Never merge silently (spec 8): ask the learner what to do.
   211→      setDevicePrompt({ userId: user.id, userName: user.name });
   212→      setAccountUser(user);
   213→      void dbMetaSet('account-user-id', user.id);
   214→      return;
   215→    }
   216→    recordAccountId(user.id);
   217→    setAccountUser(user);
   218→    void dbMetaSet('account-user-id', user.id);
   219→    toast({ title: tr('account.welcome'), description: tr('account.signedInToast') });
   220→  };
   221→
   222→  const handleSignedOut = () => {
   223→    setAccountUser(null);
   224→    void dbMetaSet('account-user-id', '');
   225→    toast({ title: tr('account.signedOutToast') });
   226→  };
   227→
   228→  const handleDeleted = () => {
   229→    setAccountUser(null);
   230→    try {
   231→      localStorage.removeItem('lumio-account-last');
   232→    } catch {
   233→      /* private mode */
   234→    }
   235→    void dbMetaSet('account-user-id', '');
   236→    toast({ title: tr('account.deletedToast') });
   237→  };
   238→
   239→  const accountSlot = (

   244→      devOutbox={devOutbox}
   245→      onSignedIn={handleSignedIn}
   246→      onSignedOut={handleSignedOut}
   247→      onDeleted={handleDeleted}
   248→    />
   249→  );
   250→
   251→  // chunk 5B: when a chapter is open in Reader OR the Quiz the panel grounds on
   252→  // THAT chapter; otherwise StudyBuddy shows a chapter select (chapter = null).
   253→  const studyBuddyChapter: BuddyChapter | null =
   254→    state?.profile?.band && (reader !== null || quiz !== null)
   255→      ? (() => {
   256→          const idx = reader ?? quiz ?? 0;
   257→          const chapters = BANDS[state.profile!.band].chapters;
   258→          const ch = chapters[Math.min(Math.max(0, idx), chapters.length - 1)];
   259→          return ch
   260→            ? { id: `${state.profile!.band}#${idx}`, title: ch.title, subject: ch.subject, chapterIndex: idx }
   261→            : null;
   262→        })()
   263→      : null;
   264→  const aiDemo = aiMode === 'fake';
   265→
   266→  // ---- chunk 5-AI-0: honest AI status chip (header) — derived above all effects ----
   267→  const aiStatusChip = (
   268→    <button
   269→      type="button"
   270→      onClick={() => setScreen('settings')}
   271→      title={
   272→        aiMode === 'zai' || aiMode === 'openai-compatible'
   273→          ? tr('ai.chipRealTitle')
   274→          : aiMode === 'fake'
   275→            ? tr('ai.chipDemoTitle')

   279→        'hidden items-center gap-1.5 rounded-full px-2.5 py-1 text-xs font-semibold sm:inline-flex min-h-[32px] ' +
   280→        (aiMode === 'zai' || aiMode === 'openai-compatible'
   281→          ? 'bg-primary/10 text-primary'
   282→          : aiMode === 'fake'
   283→            ? 'bg-gold-soft'
   284→            : 'bg-muted text-muted-foreground')
   285→      }
   286→      style={aiMode === 'fake' ? { color: 'var(--gold)' } : undefined}
   287→      aria-label={aiMode === 'zai' || aiMode === 'openai-compatible'
   288→        ? tr('ai.chipRealTitle')
   289→        : aiMode === 'fake'
   290→          ? tr('ai.chipDemoTitle')

   293→      <Sparkles className="size-3.5" aria-hidden="true" />
   294→      {aiMode === 'zai' || aiMode === 'openai-compatible'

   304→  const syllabusAvailable = !!syllabusAll;
   305→  const syllabusSubjects = (syllabusAll ?? []).filter(

   309→  const acceptedSyllabusPlan: AcceptedPlan | null = state?.syllabus.plans[todayKey] ?? null;
   310→  const dueCardsCount = Object.values(state?.srs ?? {}).filter((c) => c.due <= todayKey).length;
   311→
   312→  /** Slots for today (from the existing planner, overrides respected). */
   313→  const syllabusSlots = state?.plan ? planDayWithOverrides(state.plan, state.planOverrides[todayKey]).slots : [];
   314→
   315→  /** Nearest future exam date the learner gave us — one date for all subjects. */
   316→  const syllabusExamDate = (() => {

   323→
   324→  const examDatesForSyllabus: Record<string, string | undefined> = {};

   326→
   327→  /** Topics already planned or marked today are skipped by the planner. */

   331→      .filter(([, rec]) => rec.updatedOn === todayKey)

   347→
   348→  const deterministicSyllabusPlan: SyllabusPlanT = buildSyllabusPlanToday();
   349→
   350→  // service-worker update prompt (chunk 2): never auto-swaps mid-session

   355→      if (reg?.waiting && navigator.serviceWorker.controller) setSwWaiting(reg.waiting);
   356→    };
   357→    navigator.serviceWorker.register('/sw.js')
   358→      .then((r) => {
   359→        reg = r;
   360→        findWaiting();
   361→        r.addEventListener('updatefound', () => {
   362→          const nw = r.installing;
   363→          nw?.addEventListener('statechange', () => {
   364→            if (nw.state === 'installed') findWaiting();
   365→          });
   366→        });
   367→      })
   368→      .catch(() => { /* offline/PWA unsupported — the app still works */ });
   369→  }, []);
   370→
   371→  /**
   372→   * The one mutation path: append an event, reapply the projection, mirror the
   373→   * snapshot. Derived values (XP, best scores, streak, achievements) come out
   374→   * of the projection only — never set by callers (5.4).
   375→   */
   376→  const dispatch = useCallback((e: LumioEvent) => {
   377→    const prev = stateRef.current;
   378→    if (!prev) return;
   379→    const zone = prev.profile?.tz ?? DEFAULT_TZ;
   380→    const next = projectState(applyEvent(prev, e), new Date(), zone);
   381→    stateRef.current = next;
   382→    setState(next);
   383→    const res = saveState(next);
   384→    setSaveError(res.ok ? undefined : res.error);
   385→    // Streak Saver notifications (chunk 4C): earned / bought / used.
   386→    const prevOwned = prev.freeze.owned;
   387→    const nextOwned = next.freeze.owned;
   388→    if (nextOwned > prevOwned) {
   389→      toast({
   390→        title: e.type === 'saver-bought' ? 'Streak Saver purchased' : 'Streak Saver earned!',
   391→        description:
   392→          e.type === 'saver-bought'
   393→            ? 'It will protect your streak if a day is missed.'
   394→            : 'Great performance — your streak just got an extra life.',
   395→      });
   396→    } else if (nextOwned < prevOwned) {
   397→      toast({
   398→        title: 'Streak Saver used',
   399→        description: 'Your streak stays alive. Keep learning to earn more.',
   400→      });
   401→    }
   402→    void appendEvent(e, zone).catch(() => {
   403→      // event log unavailable (private mode) — the snapshot mirror keeps the
   404→      // session working; the learner sees an honest error if saving failed.
   405→    });
   406→  }, []);
   407→
   408→  // ---- mutations ----
   409→  const finishFirstRun = (r: FirstRunResult) => {
   410→    dispatch({ type: 'onboarding-finished', profile: r.profile, plan: r.plan });
   411→    setEditProfile(false);
   412→    setScreen('home');
   413→  };
   414→
   415→  const saveDraft = (draft: WizardDraft | null) => {
   416→    dispatch({ type: 'draft-saved', draft: draft as unknown as Record<string, unknown> | null });
   417→  };
   418→
   419→  const patchSettings = (patch: Partial<Settings>) => {
   420→    dispatch({ type: 'settings-changed', patch });
   421→  };
   422→
   423→  const patchPlan = (patch: Partial<NonNullable<LumioState['plan']>>) => {
   424→    if (!state?.plan) return;
   425→    dispatch({ type: 'plan-changed', plan: { ...state.plan, ...patch } });
   426→  };
   427→
   428→  const markPlanReviewed = () => {
   429→    if (!state) return;
   430→    const today = dayKey(new Date(), tz);
   431→    if (state.planReviews[today]) return;
   432→    // The activity makes the daily quest claimable; the Claim button grants the XP.
   433→    dispatch({ type: 'plan-reviewed', date: today });
   434→  };
   435→
   436→  const claimQuest = (questId: string, periodKey: string, xp: number) => {
   437→    if (!state) return;
   438→    if (state.quests.some((c) => c.questId === questId && c.periodKey === periodKey)) return;
   439→    dispatch({ type: 'quest-claimed', questId, periodKey, xp, date: dayKey(new Date(), tz) });
   440→  };
   441→
   442→  const finishQuiz = (chapterIndex: number, outcome: QuizOutcome) => {
   443→    dispatch({
   444→      type: 'quiz-attempted',
   445→      chapterId: `${state?.profile?.band ?? 'sec'}#${chapterIndex}`,
   446→      score: outcome.score,
   447→      nq: outcome.nq,
   448→      wrongIds: outcome.wrongIds,
   449→      theme,
   450→      date: dayKey(new Date(), tz),
   451→    });
   452→    setQuiz(null);
   453→    setReader(null);
   454→    setScreen('home');
   455→  };
   456→
   457→  // ---- chunk 5-AI feature 2: AI fresh-quiz completion. XP flows through the
   458→  // deterministic improvement rule with the daily cap + per-chapter best
   459→  // enforced in applyEvent — AI output itself never grants anything.
   460→  const finishQuizAi = (chapterIndex: number, r: { quizHash: string; score: number; nq: number; wrongIds: string[] }) => {
   461→    dispatch({
   462→      type: 'ai-quiz-attempted',
   463→      chapterRef: `${state?.profile?.band ?? 'sec'}#${chapterIndex}`,
   464→      quizHash: r.quizHash,
   465→      score: r.score,
   466→      nq: r.nq,
   467→      wrongIds: r.wrongIds,
   468→      theme,
   469→      date: dayKey(new Date(), tz),
   470→    });
   471→    setQuiz(null);
   472→    setReader(null);
   473→    setScreen('home');
   474→  };
   475→
   476→  const flashcardsReviewed = (chapterIndex: number) => {
   477→    dispatch({ type: 'flashcards-reviewed', chapterId: `${state?.profile?.band ?? 'sec'}#${chapterIndex}`, date: dayKey(new Date(), tz) });
   478→  };
   479→
   480→  // ---- chunk 3: games, labs and the practice engine ----
   481→  const winGame = (gameId: string) => {
   482→    const prev = stateRef.current;
   483→    const firstClear = !prev?.xpLedger.some((x) => x.source === 'game' && x.ref === gameId);
   484→    dispatch({

   626→        if (found) {

   631→            }

   638→          slotIndex: item.slotIndex,
   639→          activity: item.activity,

   642→          flags: rec?.flags ?? [],

   683→          seen.add(id);
   684→        }
   685→        return true;
   686→      })

   688→        slotIndex: it.slotIndex,
   689→        subjectId: slotSubjectOf.get(it.slotIndex) ?? 'maths',
   690→        topicIds: it.topicIds,
   691→        topicTitles: it.topicIds.map((id) => titleOf.get(id) ?? id),
   692→        activity: (['learn', 'practise', 'revise', 'recall'].includes(it.activity) ? it.activity : 'learn') as 'learn' | 'practise' | 'revise' | 'recall',

   695→        reason: it.reason.slice(0, 160),
   696→      }));
