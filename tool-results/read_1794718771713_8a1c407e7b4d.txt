   128→   128→// Web Worker source (JavaScript exercises). Built at runtime from a Blob URL
   129→   129→// so the worker is fully isolated: no DOM, no app state, hard to escape.
   130→   130→// Protocol — in: WorkerRequest, out: WorkerResponse.

   132→   132→
   133→   133→const WORKER_SOURCE = `
   134→   134→self.onmessage = function (e) {
   135→   135→  var t0 = Date.now();
   136→   136→  var data = e.data || {};
   137→   137→  var code = String(data.code || '');
   138→   138→  var fnName = String(data.fnName || '');
   139→   139→  var tests = Array.isArray(data.tests) ? data.tests : [];
   140→   140→
   141→   141→  function fmt(v) {
   142→   142→    try {
   143→   143→      var s = JSON.stringify(v);
   144→   144→      return s === undefined ? String(v) : s;
   145→   145→    } catch (err) {
   146→   146→      return String(v);
   147→   147→    }
   148→   148→  }
   149→   149→
   150→   150→  function deepEq(a, b) {
   151→   151→    if (a === b) return true;
   152→   152→    if (typeof a === 'number' && typeof b === 'number' && Number.isNaN(a) && Number.isNaN(b)) return true;
   153→   153→    if (Array.isArray(a) && Array.isArray(b)) {
   154→   154→      if (a.length !== b.length) return false;
   155→   155→      for (var i = 0; i < a.length; i++) { if (!deepEq(a[i], b[i])) return false; }
   156→   156→      return true;
   157→   157→    }
   158→   158→    if (a && b && typeof a === 'object' && typeof b === 'object') {
   159→   159→      var ka = Object.keys(a).sort(), kb = Object.keys(b).sort();
   160→   160→      if (ka.length !== kb.length) return false;
   161→   161→      for (var j = 0; j < ka.length; j++) {
   162→   162→        if (ka[j] !== kb[j]) return false;
   163→   163→        if (!deepEq(a[ka[j]], b[ka[j]])) return false;
   164→   164→      }
   165→   165→      return true;
   166→   166→    }
   167→   167→    return false;
   168→   168→  }
   169→   169→
   170→   170→  var fn;
   171→   171→  try {
   172→   172→    fn = new Function('return (' + code + ')')();
   173→   173→  } catch (e1) {
   174→   174→    fn = undefined;
   175→   175→  }
   176→   176→  if (typeof fn !== 'function') {
   177→   177→    // Fallback for multi-statement code: evaluate it in an IIFE and pick up
   178→   178→    // the function by name (fnName always comes from our own exercise data).
   179→   179→    try {
   180→   180→      fn = eval('(function(){' + code + '\\n;return typeof ' + fnName + ' === "function" ? ' + fnName + ' : undefined;})()');
   181→   181→    } catch (e2) {
   182→   182→      fn = undefined;
   183→   183→    }
   184→   184→  }
   185→   185→
   186→   186→  var results = [];
   187→   187→  if (typeof fn !== 'function') {
   188→   188→    var msg = 'Your code could not be turned into a callable function. Check for syntax errors.';
   189→   189→    for (var t = 0; t < tests.length; t++) {
   190→   190→      results.push({ pass: false, got: '—', expected: fmt(tests[t][1]), error: msg });
   191→   191→    }
   192→   192→  } else {
   193→   193→    for (var k = 0; k < tests.length; k++) {
   194→   194→      var args = tests[k][0];
   195→   195→      var expected = tests[k][1];
   196→   196→      try {
   197→   197→        var got = fn.apply(null, args);
   198→   198→        results.push({ pass: deepEq(got, expected), got: fmt(got), expected: fmt(expected) });
   199→   199→      } catch (err) {
   200→   200→        results.push({
   201→   201→          pass: false,
   202→   202→          got: 'threw an error',
   203→   203→          expected: fmt(expected),
   204→   204→          error: String(err && err.message ? err.message : err),
   205→   205→        });
   206→   206→      }
   207→   207→    }
   208→   208→  }
   209→   209→
   210→   210→  self.postMessage({ results: results, durationMs: Date.now() - t0 });
   211→   211→};
   212→   212→`;

   229→   229→    '    except Exception as _e:',

   239→   239→function isTestResult(v: unknown): v is TestResult {
   240→   240→  if (typeof v !== 'object' || v === null) return false;
   241→   241→  const o = v as Record<string, unknown>;
   242→   242→  if (typeof o.pass !== 'boolean') return false;
   243→   243→  if (typeof o.got !== 'string' || typeof o.expected !== 'string') return false;
   244→   244→  if (o.error !== undefined && typeof o.error !== 'string') return false;
   245→   245→  return true;
   246→   246→}
   247→   247→
   248→   248→function isWorkerResponse(v: unknown): v is WorkerResponse {
   249→   249→  if (typeof v !== 'object' || v === null) return false;
   250→   250→  const o = v as Record<string, unknown>;
   251→   251→  return (
   252→   252→    Array.isArray(o.results) &&
   253→   253→    o.results.every(isTestResult) &&
   254→   254→    typeof o.durationMs === 'number'
   255→   255→  );
   256→   256→}

   274→   274→export function CodingLabGame(props: GameProps) {
   275→   275→  // ---- plain state ----
   276→   276→  const [selectedIndex, setSelectedIndex] = useState(0);
   277→   277→  // Saved drafts restore via a lazy initialiser (no setState in effects).
   278→   278→  // This screen only mounts client-side after the app's hydration gate, so
   279→   279→  // localStorage is safe to read during the first render.
   280→   280→  function loadDrafts(): Record<string, string> {
   281→   281→    try {
   282→   282→      const raw = window.localStorage.getItem(DRAFTS_KEY);
   283→   283→      if (!raw) return {};
   284→   284→      const parsed: unknown = JSON.parse(raw);
   285→   285→      if (parsed && typeof parsed === 'object' && !Array.isArray(parsed)) {
   286→   286→        return parsed as Record<string, string>;
   287→   287→      }
   288→   288→    } catch {
   289→   289→      // Corrupt or missing storage: start fresh with starters. Honest default.
   290→   290→    }
   291→   291→    return {};
   292→   292→  }
   293→   293→  const [drafts, setDrafts] = useState<Record<string, string>>(loadDrafts);
   294→   294→  const [completed, setCompleted] = useState<Set<string>>(new Set());
   295→   295→  const [hintsShown, setHintsShown] = useState(0);
   296→   296→  const [run, setRun] = useState<RunState>({ kind: 'idle' });

   298→   298→  const [celebrate, setCelebrate] = useState<string | null>(null);
   299→   299→  const [pyStatus, setPyStatus] = useState<PyStatus>('idle');
   300→   300→

   302→   302→  const workerRef = useRef<Worker | null>(null);
   303→   303→  const jsTimerRef = useRef<number | null>(null);
   304→   304→  const pyTimerRef = useRef<number | null>(null);
   305→   305→  const pyodideRef = useRef<PyodideInterface | null>(null);
   306→   306→  const pyLoadingRef = useRef(false); // guards double-start across re-renders
   307→   307→  const pyUnmountedRef = useRef(false);
   308→   308→
   309→   309→  // ---- derivations (consts ABOVE all effects, per project rule) ----
   310→   310→  const ex = EXERCISES[selectedIndex];
   311→   311→  const isPython = ex.lang === 'py';
   312→   312→  const code = drafts[ex.id] ?? ex.starter;
   313→   313→  const hints = HINTS[ex.id] ?? [];
   314→   314→  const passCount =
   315→   315→    run.kind === 'done' ? run.results.filter((r) => r.pass).length : 0;
   316→   316→  const summary =
   317→   317→    run.kind === 'done'
   318→   318→      ? `${passCount} of ${run.results.length} tests pass`
   319→   319→      : run.kind === 'running'
   320→   320→        ? 'Running…'

   329→   329→  useEffect(() => {
   330→   330→    if (Object.keys(drafts).length === 0) return;

   338→   338→  // Lazily load Pyodide ONLY when a Python exercise is first opened. The
   339→   339→  // start is guarded by a ref (not pyStatus state) so a re-render can never
   340→   340→  // cancel an in-flight load; the status change runs in a microtask so no
   341→   341→  // setState happens synchronously inside the effect body.
   342→   342→  useEffect(() => {
   343→   343→    if (!isPython || pyLoadingRef.current) return;
   344→   344→    pyLoadingRef.current = true;
   345→   345→    queueMicrotask(() => {
   346→   346→      setPyStatus('loading');
   347→   347→
   348→   348→      const finish = () => {
   349→   349→        const load = getPyodideLoader();
   350→   350→        if (typeof load !== 'function') {
   351→   351→          setPyStatus('failed');
   352→   352→          return;
   353→   353→        }
   354→   354→        load({ indexURL: PYODIDE_INDEX_URL })
   355→   355→          .then((py) => {
   356→   356→            if (pyUnmountedRef.current) return;
   357→   357→            pyodideRef.current = py;
   358→   358→            setPyStatus('ready');
   359→   359→          })
   360→   360→          .catch(() => {
   361→   361→            if (pyUnmountedRef.current) return;
   362→   362→            setPyStatus('failed');
   363→   363→          });
   364→   364→      };

   366→   366→      const existing = document.querySelector(
   367→   367→        'script[data-lumio-pyodide]'

   369→   369→      if (existing) {
   370→   370→        if (getPyodideLoader()) {
   371→   371→          finish();
   372→   372→        } else {
   373→   373→          existing.addEventListener('load', finish, { once: true });
   374→   374→          existing.addEventListener(
   375→   375→            'error',
   376→   376→            () => {
   377→   377→              if (!pyUnmountedRef.current) setPyStatus('failed');
   378→   378→            },
   379→   379→            { once: true }
   380→   380→          );
   381→   381→        }
   382→   382→      } else {
   383→   383→        const script = document.createElement('script');
   384→   384→        script.src = PYODIDE_SCRIPT_URL;
   385→   385→        script.async = true;
   386→   386→        script.dataset.lumioPyodide = 'true';
   387→   387→        script.addEventListener('load', finish, { once: true });
   388→   388→        script.addEventListener(
   389→   389→          'error',
   390→   390→          () => {
   391→   391→            if (!pyUnmountedRef.current) setPyStatus('failed');
   392→   392→          },
   393→   393→          { once: true }
   394→   394→        );
   395→   395→        document.head.appendChild(script);
   396→   396→      }
   397→   397→    });
   398→   398→  }, [isPython]);

   400→   400→  // Unmount cleanup: terminate any live worker and clear every timer.

   402→   402→    return () => {
   403→   403→      pyUnmountedRef.current = true;
   404→   404→      if (workerRef.current) {
   405→   405→        workerRef.current.terminate();
   406→   406→        workerRef.current = null;
   407→   407→      }
   408→   408→      if (jsTimerRef.current !== null) window.clearTimeout(jsTimerRef.current);
   409→   409→      if (pyTimerRef.current !== null) window.clearTimeout(pyTimerRef.current);
   410→   410→    };
   411→   411→  }, []);

   440→   440→  function finishRun(results: TestResult[], durationMs: number) {
   441→   441→    setRun({ kind: 'done', results, durationMs });
   442→   442→    const allPass = results.length > 0 && results.every((r) => r.pass);
   443→   443→    if (allPass && !completed.has(ex.id)) {

   446→   446→        next.add(ex.id);

   449→   449→      setCelebrate(
   450→   450→        `Brilliant — every test passes for “${ex.title}”. That is real code you wrote.`
   451→   451→      );
   452→   452→      props.onWin();
   453→   453→    } else if (allPass) {
   454→   454→      setCelebrate(`All tests still pass for “${ex.title}” — nice and clean.`);
   455→   455→    } else {
   456→   456→      setCelebrate(null);
   457→   457→    }
   458→   458→  }

   460→   460→  function runJsTests() {
   461→   461→    if (run.kind === 'running') return;
   462→   462→    if (typeof Worker === 'undefined') {
   463→   463→      setRun({
   464→   464→        kind: 'error',
   465→   465→        message:
   466→   466→          'This browser does not support Web Workers, so I cannot safely run your JavaScript here. Nothing was faked — the tests simply cannot run in this browser.',
   467→   467→      });
   468→   468→      return;
   469→   469→    }
   470→   470→    setRun({ kind: 'running' });
   471→   471→    setCelebrate(null);
   472→   472→    setPrintOutput(null);
   473→   473→
   474→   474→    // Fresh worker per run: created lazily, terminated after use, no leaks.
   475→   475→    const blob = new Blob([WORKER_SOURCE], { type: 'text/javascript' });
   476→   476→    const url = URL.createObjectURL(blob);
   477→   477→    const worker = new Worker(url);
   478→   478→    URL.revokeObjectURL(url); // the worker has already taken its reference
   479→   479→    workerRef.current = worker;
   480→   480→
   481→   481→    if (jsTimerRef.current !== null) window.clearTimeout(jsTimerRef.current);
   482→   482→    jsTimerRef.current = window.setTimeout(() => {
   483→   483→      worker.terminate();
   484→   484→      workerRef.current = null;
   485→   485→      jsTimerRef.current = null;
   486→   486→      setRun({
   487→   487→        kind: 'error',
   488→   488→        message:
   489→   489→          'Your code ran too long (over 2 seconds) — check for an infinite loop.',
   490→   490→      });

   492→   492→

   494→   494→      if (jsTimerRef.current !== null) window.clearTimeout(jsTimerRef.current);
   495→   495→      jsTimerRef.current = null;
   496→   496→      worker.terminate();
   497→   497→      workerRef.current = null;
   498→   498→      if (!isWorkerResponse(ev.data)) {
   499→   499→        setRun({
   500→   500→          kind: 'error',
   501→   501→          message: 'The test runner returned something unexpected. Please try again.',
   502→   502→        });
   503→   503→        return;
   504→   504→      }
   505→   505→      finishRun(ev.data.results, ev.data.durationMs);
   506→   506→    };
   507→   507→    worker.onerror = (ev: ErrorEvent) => {
   508→   508→      if (jsTimerRef.current !== null) window.clearTimeout(jsTimerRef.current);
   509→   509→      jsTimerRef.current = null;
   510→   510→      worker.terminate();
   511→   511→      workerRef.current = null;
   512→   512→      setRun({
   513→   513→        kind: 'error',
   514→   514→        message: `Your code crashed while running: ${ev.message || 'unknown error'}`,
   515→   515→      });
   516→   516→    };
   517→   517→
   518→   518→    const request: WorkerRequest = {
   519→   519→      code,
   520→   520→      fnName: ex.id,
   521→   521→      tests: ex.tests,
   522→   522→    };
   523→   523→    worker.postMessage(request);
   524→   524→  }

   526→   526→  async function runPythonTests() {
   527→   527→    if (run.kind === 'running') return;
   528→   528→    if (pyStatus === 'failed') {
   529→   529→      setRun({ kind: 'error', message: PY_FAIL_MESSAGE });
   530→   530→      return;
   531→   531→    }
   532→   532→    const py = pyodideRef.current;
   533→   533→    if (pyStatus !== 'ready' || !py) {
   534→   534→      setRun({
   535→   535→        kind: 'error',
   536→   536→        message:
   537→   537→          'Python is still downloading its runtime (Pyodide). Give it a few seconds, then press “Run tests” again.',
   538→   538→      });
   539→   539→      return;
   540→   540→    }
   541→   541→
   542→   542→    setRun({ kind: 'running' });
   543→   543→    setCelebrate(null);
   544→   544→    setPrintOutput(null);
   545→   545→
   546→   546→    if (pyTimerRef.current !== null) window.clearTimeout(pyTimerRef.current);
   547→   547→
   548→   548→    // Promise.race: the honest 3-second limit on the Python run.
   549→   549→    const exec = (async () => {
   550→   550→      const outs: string[] = [];
   551→   551→      py.setStdout({ batched: (line: string) => { outs.push(line); } });
   552→   552→      // 1) The learner's code defines the function (a syntax error here is
   553→   553→      //    reported honestly, never faked).
   554→   554→      py.runPython(code);
   555→   555→      // 2) The harness calls the function with each test's JSON-serialised
   556→   556→      //    args and compares with == inside Python.
   557→   557→      py.globals.set('_lumio_tests_json', JSON.stringify(ex.tests));
   558→   558→      const raw = await py.runPythonAsync(pythonHarness(ex.id));
   559→   559→      if (typeof raw !== 'string') {
   560→   560→        throw new Error('the test harness returned something unexpected');
   561→   561→      }
   562→   562→      const parsed: unknown = JSON.parse(raw);
   563→   563→      if (!Array.isArray(parsed) || !parsed.every(isTestResult)) {
   564→   564→        throw new Error('the test results were not in the expected shape');
   565→   565→      }
   566→   566→      if (outs.length > 0) setPrintOutput(outs.join('\n'));
   567→   567→      return parsed as TestResult[];
   568→   568→    })();
   569→   569→
   570→   570→    const timeout = new Promise<never>((_, reject) => {
   571→   571→      pyTimerRef.current = window.setTimeout(
   572→   572→        () => reject(new Error(PY_TIMEOUT_MESSAGE)),
   573→   573→        PY_TIMEOUT_MS
   574→   574→      );
   575→   575→    });
   576→   576→
   577→   577→    try {
   578→   578→      const results = await Promise.race([exec, timeout]);
   579→   579→      if (pyTimerRef.current !== null) window.clearTimeout(pyTimerRef.current);
   580→   580→      pyTimerRef.current = null;
   581→   581→      finishRun(results, 0);
   582→   582→    } catch (err) {
   583→   583→      if (pyTimerRef.current !== null) window.clearTimeout(pyTimerRef.current);
   584→   584→      pyTimerRef.current = null;
   585→   585→      setRun({
   586→   586→        kind: 'error',
   587→   587→        message: `Python could not run your code: ${
   588→   588→          err instanceof Error ? err.message : String(err)
   589→   589→        }`,
   590→   590→      });
   591→   591→    }
   592→   592→  }
