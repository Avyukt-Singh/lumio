    44→    44→/**

    62→    62→  const [accountAvail, setAccountAvail] = useState<'unknown' | 'ok' | 'unavailable'>('unknown');

    97→    97→      if (!alive) return;

    99→    99→        const events = await loadEventsAndSeed();
   100→   100→        const projected = replayEvents(events, new Date(), events.length && (events[events.length - 1] as { tz?: string }).tz ? ((events[events.length - 1] as unknown as { tz: string }).tz) : DEFAULT_TZ);
   101→   101→        if (alive) {
   102→   102→          setState(projected);
   103→   103→          saveState(projected);
   104→   104→        }

   106→   106→        const snapshot = loadState();

   133→   133→

   137→   137→    const root = document.documentElement;
   138→   138→    root.classList.remove('lang-hi', 'lang-hinglish', 'lang-en');

   166→   166→      } else {
   167→   167→        setAccountUser(null);
   168→   168→      }

   177→   177→      /* private mode */
   178→   178→    }
   179→   179→  };

   184→   184→        return localStorage.getItem('lumio-account-last');

   186→   186→        return null;
   187→   187→      }
   188→   188→    })();
   189→   189→    const hasLocalProgress = !!stateRef.current?.profile;

   194→   194→      void dbMetaSet('account-user-id', user.id);

   196→   196→    }
   197→   197→    recordAccountId(user.id);
   198→   198→    setAccountUser(user);
   199→   199→    void dbMetaSet('account-user-id', user.id);

   202→   202→

   242→   242→        })()
   243→   243→      : null;

   245→   245→  const aiEnabled = accountUser?.aiEnabled === true;

   248→   248→  // service-worker update prompt (chunk 2): never auto-swaps mid-session

   264→   264→        });
   265→   265→      })
   266→   266→      .catch(() => { /* offline/PWA unsupported — the app still works */ });
   267→   267→  }, []);
   268→   268→
   269→   269→  /**
   270→   270→   * The one mutation path: append an event, reapply the projection, mirror the
   271→   271→   * snapshot. Derived values (XP, best scores, streak, achievements) come out
   272→   272→   * of the projection only — never set by callers (5.4).
   273→   273→   */
   274→   274→  const dispatch = useCallback((e: LumioEvent) => {
   275→   275→    const prev = stateRef.current;
   276→   276→    if (!prev) return;
   277→   277→    const zone = prev.profile?.tz ?? DEFAULT_TZ;
   278→   278→    const next = projectState(applyEvent(prev, e), new Date(), zone);
   279→   279→    stateRef.current = next;
   280→   280→    setState(next);
   281→   281→    const res = saveState(next);
   282→   282→    setSaveError(res.ok ? undefined : res.error);
   283→   283→    // Streak Saver notifications (chunk 4C): earned / bought / used.
   284→   284→    const prevOwned = prev.freeze.owned;
   285→   285→    const nextOwned = next.freeze.owned;
   286→   286→    if (nextOwned > prevOwned) {
   287→   287→      toast({
   288→   288→        title: e.type === 'saver-bought' ? 'Streak Saver purchased' : 'Streak Saver earned!',
   289→   289→        description:
   290→   290→          e.type === 'saver-bought'
   291→   291→            ? 'It will protect your streak if a day is missed.'
   292→   292→            : 'Great performance — your streak just got an extra life.',
   293→   293→      });
   294→   294→    } else if (nextOwned < prevOwned) {
   295→   295→      toast({
   296→   296→        title: 'Streak Saver used',
   297→   297→        description: 'Your streak stays alive. Keep learning to earn more.',
   298→   298→      });
   299→   299→    }
   300→   300→    void appendEvent(e, zone).catch(() => {
   301→   301→      // event log unavailable (private mode) — the snapshot mirror keeps the
   302→   302→      // session working; the learner sees an honest error if saving failed.
   303→   303→    });
   304→   304→  }, []);
   305→   305→
   306→   306→  // ---- mutations ----

   364→   364→      type: 'game-won',

   368→   368→      lab: LAB_GAME_IDS.has(gameId),

   370→   370→    });
   371→   371→  };
   372→   372→  const reviewCard = (cardId: string, correct: boolean) => {

   374→   374→  };

   378→   378→  const finishLegendary = () => {

   380→   380→  };
   381→   381→  const tagMistake = (questionId: string, tag: MistakeTag | null) => {

   384→   384→
   385→   385→  // ---- chunk 5B: AI practice-only saves (kind 'notes' | 'practice') ----
   386→   386→  // Never touches XP/streak/mastery — the projection keeps the record in

   388→   388→  const saveAiContent = (

   391→   391→    date: string,

   394→   394→  ) => {

   396→   396→  };
   397→   397→  const saveAiNotes = (args: {
   398→   398→    id: string;
   399→   399→    kind: 'notes';
   400→   400→    date: string;
   401→   401→    payload: NotesAiResult;
   402→   402→    label: string;
   403→   403→  }) => {

   405→   405→  };
   406→   406→
   407→   407→  // ---- chunk 4A: focus suite, mocks, streak freezes, reschedule ----
   408→   408→  const finishFocusBlock = (mins: number) => {
   409→   409→    dispatch({ type: 'focus-block-done', date: dayKey(new Date(), tz), mins });
   410→   410→  };
   411→   411→  const recordMock = (mock: { date: string; subject: string; score: number; total: number; weakChapters: number[] }) => {
   412→   412→    dispatch({ type: 'mock-recorded', mock });
   413→   413→  };
   414→   414→  const useFreeze = (date: string) => {

   417→   417→  const rescheduleSession = (date: string, index: number, startMin: number | null) => {
