    17→export function dayKey(d: Date, tz: string): string {
    18→  const fmt = new Intl.DateTimeFormat('en-CA', {
    19→    timeZone: tz,
    20→    year: 'numeric',
    21→    month: '2-digit',
    22→    day: '2-digit',
    23→  });
    24→  return fmt.format(d); // en-CA gives yyyy-mm-dd
    25→}

    33→export function weekdayOf(d: Date, tz: string): number {
    34→  // 0 = Sunday ... 6 = Saturday, computed from the learner-local date
    35→  const wd = new Intl.DateTimeFormat('en-US', { timeZone: tz, weekday: 'short' }).format(d);
    36→  return ['Sun', 'Mon', 'Tue', 'Wed', 'Thu', 'Fri', 'Sat'].indexOf(wd);

    39→/** Monday-start ISO week key like 2025-W45 (5.6: weeks start on Monday). */
    40→export function weekKey(d: Date, tz: string): string {
    41→  const dateStr = dayKey(d, tz);
    42→  const dt = new Date(dateStr + 'T12:00:00Z');
    43→  const day = (dt.getUTCDay() + 6) % 7; // Mon=0
    44→  dt.setUTCDate(dt.getUTCDate() - day + 3); // shift to Thursday of this ISO week
    45→  const isoYear = dt.getUTCFullYear();
    46→  const jan4 = new Date(Date.UTC(isoYear, 0, 4));
    47→  const jan4Day = (jan4.getUTCDay() + 6) % 7;
    48→  const firstThu = new Date(jan4);
    49→  firstThu.setUTCDate(jan4.getUTCDate() - jan4Day + 3);
    50→  const weekNum = 1 + Math.round((dt.getTime() - firstThu.getTime()) / (7 * 86400000));
    51→  return `${isoYear}-W${String(Math.max(1, weekNum)).padStart(2, '0')}`;
    52→}

    58→export function addDays(dateStr: string, n: number): string {
    59→  const d = new Date(dateStr + 'T12:00:00Z');
    60→  d.setUTCDate(d.getUTCDate() + n);
    61→  return d.toISOString().slice(0, 10);

    66→export function themeAt(hour: number, sch: [number, number, number]): ThemeName {
    67→  // sch = [morningStart, eveningStart, nightStart] (defaults 06 / 17 / 19).
    68→  // Night wraps past midnight: 19:00 -> 05:59 (prototype 3.2).
    69→  const [morningStart, eveningStart, nightStart] = sch;
    70→  if (hour >= nightStart || hour < morningStart) return 'night';
    71→  if (hour >= eveningStart) return 'evening';
    72→  return 'morning';
    73→}

    85→/**
    86→ * XP for improvement rule (3.6): gain = max(0, score - previousBest) * 10,
    87→ * plus a one-time +20 on the first scored attempt with score >= 1.
    88→ */
    89→export function quizXpGain(prevBest: number, score: number, firstAttempt: boolean): number {
    90→  const improvement = Math.max(0, score - prevBest) * 10;
    91→  const firstBonus = firstAttempt && score >= 1 ? 20 : 0;
    92→  return improvement + firstBonus;

    97→/**
    98→ * Streak = consecutive learner-local days with an XP-earning activity, counting back
    99→ * from today, or from yesterday if today is not yet done. Opening the app never counts.
   100→ * A spent streak freeze (`freezeUsed[date]`) shields one missed day, keeping the chain
   101→ * alive (chunk 4A, 7.10).
   102→ */
   103→export function streakFrom(
   104→  days: string[],
   105→  today: string,
   106→  freezeUsed?: Record<string, true>
   107→): number {
   108→  const set = new Set(days);
   109→  const shield = freezeUsed ?? {};
   110→  const active = (d: string) => set.has(d) || shield[d] === true;
   111→  let cursor = active(today) ? today : addDays(today, -1);
   112→  let n = 0;
   113→  while (active(cursor)) {
   114→    n += 1;
   115→    cursor = addDays(cursor, -1);
   116→  }
   117→  return n;
   118→}

   124→  return null;

   139→/** A chapter unlocks when the previous one is completed (best >= 1). */
   140→export function chapterState(
   141→  band: Band,
   142→  index: number,
   143→  res: Record<string, ChapterResult>
   144→): ChapterState {
   145→  const nq = BANDS[band].chapters[index]?.questions.length ?? 0;
   146→  const best = bestScore(res, chapterId(band, index));
   147→  if (best >= nq && nq > 0) return 'mastered';
   148→  if (best >= 1) return 'completed';
   149→  if (index === 0) return 'available';
   150→  return bestScore(res, chapterId(band, index - 1)) >= 1 ? 'available' : 'locked';
   151→}

   153→/** Number of chapters completed (best >= 1) and mastered (best == nq). */
   154→export function chapterTally(band: Band, res: Record<string, ChapterResult>) {
   155→  const n = BANDS[band].chapters.length;
   156→  let completed = 0;
   157→  let mastered = 0;
   158→  for (let i = 0; i < n; i++) {
   159→    const st = chapterState(band, i, res);
   160→    if (st === 'completed' || st === 'mastered') completed += 1;
   161→    if (st === 'mastered') mastered += 1;
   162→  }
   163→  return { completed, mastered, total: n };
   164→}

   166→// ---------- scheduler (3.6 / 7.3 basics) ----------
   167→// Sessions are placed after school, around dinner, before bedtime, with 15-minute gaps.
   168→// Busy windows: school hours and dinner (45 minutes).
   169→
   170→export interface SessionSlot {
   171→  startMin: number;
   172→  endMin: number;
   173→  label: string;
   174→}
   175→
   176→export const GAP_MIN = 15;
   177→export const DINNER_LEN = 45;
   178→
   179→/** Free study windows for a day: after school/work and after dinner, minus commitment blocks. */
   180→function freeWindows(plan: SchedulePlan): Array<[number, number]> {
   181→  const { startMin: _s, endMin, dinnerMin, bedMin, sessionLen } = plan;
   182→  let free: Array<[number, number]> = [];
   183→  if (endMin + GAP_MIN <= dinnerMin) free.push([endMin + GAP_MIN, dinnerMin]);
   184→  if (dinnerMin + DINNER_LEN <= bedMin) free.push([dinnerMin + DINNER_LEN, bedMin]);
   185→
   186→  // subtract commitment blocks (clamped to sensible bounds) from the free windows
   187→  const blocks = (plan.busy ?? [])
   188→    .filter((c) => c.endMin > c.startMin)
   189→    .map((c) => [Math.max(0, c.startMin), Math.min(24 * 60, c.endMin)] as [number, number])
   190→    .sort((a, b) => a[0] - b[0]);
   191→  for (const [bs, be] of blocks) {
   192→    const next: Array<[number, number]> = [];
   193→    for (const [ws, we] of free) {
   194→      if (be <= ws || bs >= we) {
   195→        next.push([ws, we]); // no overlap
   196→        continue;
   197→      }
   198→      if (bs > ws) next.push([ws, Math.min(bs, we)]);
   199→      if (be < we) next.push([Math.max(be, ws), we]);
   200→    }
   201→    free = next.filter(([ws, we]) => we - ws >= Math.min(10, sessionLen));
   202→  }
   203→  return free;
   204→}
   205→
   206→/**
   207→ * Lay out `plan.sessions` study sessions for one day. Sessions go after
   208→ * school/work and after dinner, before bedtime, with 15-minute gaps. Structured
   209→ * commitment blocks (7.1 upgrade) are subtracted from the free windows —
   210→ * the planner never overlaps a commitment. Returns the slots, or a
   211→ * "day too full" verdict instead of guessing (3.3: if the day is too full say so).
   212→ */
   213→export function planDay(plan: SchedulePlan): { slots: SessionSlot[]; dropped: number } {
   214→  const { sessions, sessionLen } = plan;
   215→  const free = freeWindows(plan);
   216→
   217→  const slots: SessionSlot[] = [];
   218→  for (const [ws, we] of free) {
   219→    let t = ws;
   220→    while (slots.length < sessions && t + sessionLen <= we) {
   221→      slots.push({
   222→        startMin: t,
   223→        endMin: t + sessionLen,
   224→        label: sessions === 1 ? 'Study session' : `Session ${slots.length + 1}`,
   225→      });
   226→      t += sessionLen + GAP_MIN;
   227→    }
   228→    if (slots.length >= sessions) break;
   229→  }
   230→  return { slots, dropped: sessions - slots.length };
   231→}
   232→
   233→/**
   234→ * The day's plan with the learner's interactive reschedule overrides applied
   235→ * (chunk 4A). Overrides are validated when generated (rescheduleOptions); an
   236→ * override for a slot index that does not exist is ignored.
   237→ */
   238→export function planDayWithOverrides(
   239→  plan: SchedulePlan,
   240→  overrides?: Record<string, number>
   241→): { slots: SessionSlot[]; dropped: number } {
   242→  const base = planDay(plan);
   243→  if (!overrides || Object.keys(overrides).length === 0) return base;
   244→  const slots = base.slots.map((s, i) => {
   245→    const ov = overrides[String(i)];
   246→    if (typeof ov !== 'number') return s;
   247→    return { ...s, startMin: ov, endMin: ov + (plan.sessionLen ?? s.endMin - s.startMin) };
   248→  });
   249→  return { slots, dropped: base.dropped };
   250→}
   251→
   252→/**
   253→ * Every valid 15-minute-aligned start time for moving session `index`: inside a
   254→ * free window (school/dinner/commitments honoured), not overlapping another
   255→ * placed session, and not its own current start.
   256→ */
   257→export function rescheduleOptions(
   258→  plan: SchedulePlan,
   259→  overrides: Record<string, number> | undefined,
   260→  index: number
   261→): number[] {
   262→  const { slots } = planDay(plan);
   263→  if (index < 0 || index >= slots.length) return [];
   264→  const len = plan.sessionLen;
   265→  const others = slots
   266→    .map((s, i) => {
   267→      const ov = overrides?.[String(i)];
   268→      const start = typeof ov === 'number' ? ov : s.startMin;
   269→      return { i, startMin: start, endMin: start + len };
   270→    })
   271→    .filter((s) => s.i !== index);
   272→  const out: number[] = [];
   273→  for (const [ws, we] of freeWindows(plan)) {
   274→    for (let t = ws; t + len <= we; t += 15) {
   275→      const clash = others.some((o) => t < o.endMin && o.startMin < t + len);
   276→      if (!clash && t !== slots[index].startMin) out.push(t);
   277→    }
   278→  }
   279→  return out;
   280→}

   282→/**
   283→ * Subject pairing (7.3): even-indexed sessions carry a core subject that rotates
   284→ * day by day; odd-indexed sessions pair with the complementary subject when one
   285→ * is set (session 2 pairs with the same core as session 1, keeping the pair). */
   286→export function subjectForSession(
   287→  subjects: string[],
   288→  compSub: string | undefined,
   289→  dayIndex: number,
   290→  sessionIndex: number
   291→): string | null {
   292→  if (subjects.length === 0 && !compSub) return null;
   293→  if (sessionIndex % 2 === 1 && compSub) return compSub;
   294→  const pool = subjects.length > 0 ? subjects : compSub ? [compSub] : [];
   295→  if (pool.length === 0) return null;
   296→  const pairIndex = Math.floor(sessionIndex / 2);
   297→  return pool[(dayIndex + pairIndex) % pool.length];
   298→}

   300→/** The 7 dates Monday..Sunday of the week containing `today` (5.6: weeks start Monday). */
   301→export function weekDatesOf(today: string): string[] {
   302→  const d = new Date(today + 'T12:00:00Z');
   303→  const day = (d.getUTCDay() + 6) % 7; // Mon=0
   304→  const monday = addDays(today, -day);
   305→  return Array.from({ length: 7 }, (_, i) => addDays(monday, i));
   306→}

   308→export function minToLabel(min: number): string {
   309→  const h = Math.floor(min / 60);
   310→  const m = min % 60;
   311→  const ampm = h >= 12 ? 'PM' : 'AM';
   312→  const h12 = h % 12 === 0 ? 12 : h % 12;
   313→  return `${h12}:${String(m).padStart(2, '0')} ${ampm}`;
   314→}

   404→export function questProgress(
   405→  state: Pick<
   406→    LumioState,
   407→    'days' | 'xpLedger' | 'quests' | 'planReviews' | 'practice' | 'focusLog' | 'mocks'
   408→  >,
   409→  now: Date,
   410→  tz: string
   411→): QuestProgress[] {
   412→  const today = dayKey(now, tz);
   413→  const wk = weekKey(now, tz);
   414→  const mk = monthKey(now, tz);
   415→  const weekDays = new Set(weekDatesOf(today));
   416→  return QUESTS.map((q) => {
   417→    let progress = 0;
   418→    let target = 1;
   419→    if (q.id === 'daily-plan-review') {
   420→      progress = state.planReviews[today] ? 1 : 0;
   421→    } else if (q.id === 'daily-focus-block') {
   422→      progress = Math.min(1, state.focusLog[today]?.blocks ?? 0);
   423→    } else if (q.id === 'daily-cards') {
   424→      progress = state.practice['srs#' + today] ? 1 : 0;
   425→    } else if (q.id === 'weekly-3-days') {
   426→      target = 3;
   427→      progress = Math.min(target, daysInWeek(state.days, wk));
   428→    } else if (q.id === 'weekly-practice') {
   429→      target = 3;
   430→      const practiceDays = new Set(
   431→        state.xpLedger
   432→          .filter((e) => e.source === 'practice' && weekDays.has(e.date))
   433→          .map((e) => e.date)
   434→      );
   435→      progress = Math.min(target, practiceDays.size);
   436→    } else if (q.id === 'weekly-mock') {
   437→      progress = Math.min(1, state.mocks.filter((m) => weekDays.has(m.date)).length);
   438→    } else if (q.id === 'monthly-100xp') {
   439→      target = 100;
   440→      progress = Math.min(
   441→        target,
   442→        state.xpLedger
   443→          .filter((e) => e.date.startsWith(mk) && e.source !== 'import' && e.source !== 'quest')
   444→          .reduce((s, e) => s + Math.max(0, e.amount), 0)
   445→      );
   446→    } else if (q.id === 'monthly-8-days') {
   447→      target = 8;
   448→      progress = Math.min(target, state.days.filter((d) => d.startsWith(mk)).length);
   449→    }
   450→    const claimed = state.quests.some(
   451→      (c) => c.questId === q.id && c.periodKey === (q.period === 'daily' ? today : q.period === 'weekly' ? wk : mk)
   452→    );
   453→    return { quest: q, progress, target, claimable: progress >= target && !claimed, claimed };
   454→  });
   455→}

   600→export interface WeakTopic {
   601→  chapterIndex: number;
   602→  chapterTitle: string;
   603→  subject: string;
   604→  /** distinct missed question ids in this chapter */
   605→  misses: number;
   606→  /** chapter weight emphasises earlier fundamentals (2/3/1/2/1/2 ranking heuristic) */
   607→  weight: number;
   608→}
   609→
   610→/**
   611→ * Weak-topic ranking from `state.wrong` (stable ids `${band}#${ch}#${q}`),
   612→ * merged with the chapters the learner marked weak in self-reported mocks
   613→ * (chunk 4A — each mock marking adds one miss-like weight). Chapters are
   614→ * ranked by distinct misses, then by an importance weight that emphasises
   615→ * the chapters the study path treats as foundations.
   616→ */
   617→export function weakTopics(state: LumioState, limit = 6): WeakTopic[] {
   618→  if (!state.profile) return [];
   619→  const band = state.profile.band;
   620→  const counts = new Map<number, Set<string>>();
   621→  for (const id of Object.keys(state.wrong)) {
   622→    const [b, ch] = id.split('#');
   623→    if (b !== band) continue;
   624→    const idx = Number(ch);
   625→    if (!Number.isFinite(idx)) continue;
   626→    if (!counts.has(idx)) counts.set(idx, new Set());
   627→    counts.get(idx)!.add(id);
   628→  }
   629→  // mock-reported weak chapters merge into the same ranking
   630→  for (const mock of state.mocks ?? []) {
   631→    for (const idx of mock.weakChapters ?? []) {
   632→      if (!counts.has(idx)) counts.set(idx, new Set());
   633→      counts.get(idx)!.add(`mock#${mock.date}#${mock.subject}`);
   634→    }
   635→  }
   636→  const chapters = BANDS[band].chapters;
   637→  const weightFor = (i: number) => [2, 3, 1, 2, 1, 2][i % 6];
   638→  const rows: WeakTopic[] = [];
   639→  for (const [idx, ids] of counts) {
   640→    const ch = chapters[idx];
   641→    if (!ch) continue;
   642→    rows.push({
   643→      chapterIndex: idx,
   644→      chapterTitle: ch.title,
   645→      subject: ch.subject,
   646→      misses: ids.size,
   647→      weight: weightFor(idx),
   648→    });
   649→  }
   650→  rows.sort((a, b) => b.misses * b.weight - a.misses * a.weight || a.chapterIndex - b.chapterIndex);
   651→  return rows.slice(0, limit);
   652→}

   660→/** Flashcards due today (or overdue) grouped by chapter, for the SRS session. */
   661→export function dueCards(state: LumioState, today: string, cards: Array<{ id: string; chapterIndex: number; chapterTitle: string }>): DueCardGroup[] {
   662→  const due = cards.filter((c) => {
   663→    const s = state.srs[c.id];
   664→    return !s || s.due <= today; // never-seen cards are always due
   665→  });
   666→  const groups = new Map<number, DueCardGroup>();
   667→  for (const c of due) {
   668→    if (!groups.has(c.chapterIndex)) {
   669→      groups.set(c.chapterIndex, { chapterIndex: c.chapterIndex, chapterTitle: c.chapterTitle, cardIds: [] });
   670→    }
   671→    groups.get(c.chapterIndex)!.cardIds.push(c.id);
   672→  }
   673→  return [...groups.values()];
   674→}

   678→/**
   679→ * A Monday-start month grid (6 rows x 7 cols). Cells outside the month are
   680→ * null, so the calendar never invents neighbouring-month data.
   681→ */
   682→export function monthCells(year: number, month0: number): Array<string | null> {
   683→  const first = new Date(Date.UTC(year, month0, 1));
   684→  const lead = (first.getUTCDay() + 6) % 7; // Mon=0 blank cells
   685→  const dayCount = new Date(Date.UTC(year, month0 + 1, 0)).getUTCDate();
   686→  const cells: Array<string | null> = Array.from({ length: lead }, () => null);
   687→  for (let d = 1; d <= dayCount; d++) {
   688→    cells.push(
   689→      `${year}-${String(month0 + 1).padStart(2, '0')}-${String(d).padStart(2, '0')}`
   690→    );
   691→  }
   692→  while (cells.length % 7 !== 0) cells.push(null);
   693→  return cells;
   694→}

   696→function pad(n: number): string {
   697→  return String(n).padStart(2, '0');
   698→}

   700→/**
   701→ * Convert a learner-local wall-clock time (date + minutes) in `tz` to the true
   702→ * UTC instant. Iterative offset resolution handles DST boundaries honestly.
   703→ */
   704→export function zonedMinutesToUtc(dateStr: string, minutes: number, tz: string): Date {
   705→  const hh = Math.floor(minutes / 60) % 24;
   706→  const mm = minutes % 60;
   707→  const naive = Date.parse(`${dateStr}T${pad(hh)}:${pad(mm)}:00Z`);
   708→  let ts = naive;
   709→  const fmt = new Intl.DateTimeFormat('en-CA', {
   710→    timeZone: tz,
   711→    year: 'numeric',
   712→    month: '2-digit',
   713→    day: '2-digit',
   714→    hour: '2-digit',
   715→    minute: '2-digit',
   716→    second: '2-digit',
   717→    hour12: false,
   718→  });
   719→  for (let i = 0; i < 3; i++) {
   720→    const parts = fmt.formatToParts(new Date(ts));
   721→    const get = (t: string) => Number(parts.find((p) => p.type === t)?.value ?? 0);
   722→    const asUtc = Date.UTC(
   723→      get('year'),
   724→      get('month') - 1,
   725→      get('day'),
   726→      get('hour') % 24,
   727→      get('minute'),
   728→      get('second')
   729→    );
   730→    ts = naive - (asUtc - naive);
   731→  }
   732→  return new Date(ts);
   733→}

   735→function icsEscape(text: string): string {
   736→  return text.replace(/\\/g, '\\\\').replace(/;/g, '\\;').replace(/,/g, '\\,').replace(/\n/g, '\\n');
   737→}
   738→
   739→function icsStamp(d: Date): string {
   740→  return (
   741→    `${d.getUTCFullYear()}${pad(d.getUTCMonth() + 1)}${pad(d.getUTCDate())}T` +
   742→    `${pad(d.getUTCHours())}${pad(d.getUTCMinutes())}${pad(d.getUTCSeconds())}Z`
   743→  );
   744→}

   746→export interface IcsEvent {
   747→  /** stable unique id, e.g. `lumio-2025-11-15-0` */
   748→  uid: string;
   749→  date: string; // yyyy-mm-dd (learner-local)
   750→  startMin: number;
   751→  endMin: number;
   752→  summary: string;
   753→  description?: string;
   754→}

   756→/**
   757→ * Build an RFC-5545 VCALENDAR string. Times are emitted as UTC instants
   758→ * (converted from the learner's zone), which every calendar app accepts
   759→ * without requiring a VTIMEZONE block.
   760→ */
   761→export function buildIcs(events: IcsEvent[], tz: string, calName: string): string {
   762→  const stamp = icsStamp(new Date());
   763→  const lines: string[] = [
   764→    'BEGIN:VCALENDAR',
   765→    'VERSION:2.0',
   766→    'PRODID:-//Lumio//Study Plan//EN',
   767→    'CALSCALE:GREGORIAN',
   768→    'METHOD:PUBLISH',
   769→    `X-WR-CALNAME:${icsEscape(calName)}`,
   770→  ];
   771→  for (const ev of events) {
   772→    const start = zonedMinutesToUtc(ev.date, ev.startMin, tz);
   773→    const end = zonedMinutesToUtc(ev.date, ev.endMin, tz);
   774→    lines.push(
   775→      'BEGIN:VEVENT',
   776→      `UID:${ev.uid}@lumio.app`,
   777→      `DTSTAMP:${stamp}`,
   778→      `DTSTART:${icsStamp(start)}`,
   779→      `DTEND:${icsStamp(end)}`,
   780→      `SUMMARY:${icsEscape(ev.summary)}`,
   781→      ev.description ? `DESCRIPTION:${icsEscape(ev.description)}` : '',
   782→      'END:VEVENT'
   783→    );
   784→  }
   785→  lines.push('END:VCALENDAR');
   786→  return lines.filter(Boolean).join('\r\n') + '\r\n';
   787→}
