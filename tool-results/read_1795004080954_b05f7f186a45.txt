   113→/** Deterministic Fisher-Yates so repeats across rounds vary with the seed. */
   114→function seededShuffle<T>(items: T[], seed: number): T[] {
   115→  const arr = [...items];
   116→  let s = seed >>> 0 || 1;
   117→  const rand = () => {
   118→    s = (s * 1664525 + 1013904223) >>> 0;
   119→    return s / 4294967296;
   120→  };
   121→  for (let i = arr.length - 1; i > 0; i--) {
   122→    const j = Math.floor(rand() * (i + 1));
   123→    [arr[i], arr[j]] = [arr[j], arr[i]];
   124→  }
   125→  return arr;
   126→}

   128→/** Resolve stable question ids (`${band}#${ch}#${q}`) against the band content. */
   129→function resolveMissed(band: Band, wrong: Record<string, 1>): MissedItem[] {
   130→  const out: MissedItem[] = [];
   131→  for (const id of Object.keys(wrong)) {
   132→    const parts = id.split('#');
   133→    if (parts.length !== 3 || parts[0] !== band) continue;
   134→    const ci = Number(parts[1]);
   135→    const qi = Number(parts[2]);
   136→    if (!Number.isFinite(ci) || !Number.isFinite(qi)) continue;
   137→    const ch = BANDS[band].chapters[ci];
   138→    const q = ch?.questions[qi];
   139→    if (!ch || !q) continue;
   140→    out.push({ id, chapterTitle: ch.title, subject: ch.subject, q });
   141→  }
   142→  return out;
   143→}

   258→function FlashcardsTab({
   259→  state,
   260→  today,
   261→  band,
   262→  onCardReview,
   263→  onPracticeDone,
   264→}: {
   265→  state: LumioState;
   266→  today: string;
   267→  band: Band;
   268→  onCardReview: (cardId: string, correct: boolean) => void;
   269→  onPracticeDone: (kind: PracticeKind) => void;
   270→}) {
   271→  const deck = BANDS[band].chapters.flatMap((ch, i) => flashcardsFor(band, i, ch));
   272→  const due = deck.filter((c) => {
   273→    const s = state.srs[c.id];
   274→    return !s || s.due <= today;
   275→  });
   276→  const boxCounts = [1, 2, 3, 4, 5].map((b) => deck.filter((c) => state.srs[c.id]?.box === b).length);
   277→  const unseen = deck.filter((c) => !state.srs[c.id]).length;
   278→  const nextDue = deck.reduce<string | null>((min, c) => {
   279→    const d = state.srs[c.id]?.due;
   280→    if (d && d > today && (min === null || d < min)) return d;
   281→    return min;
   282→  }, null);
   283→  const collected = practiceDoneToday(state, 'srs', today);
   284→
   285→  const [queue, setQueue] = useState<Flashcard[] | null>(null);
   286→  const [pos, setPos] = useState(0);
   287→  const [flipped, setFlipped] = useState(false);
   288→  const [reviewed, setReviewed] = useState(0);
   289→  const finishedRef = useRef(false);
   290→
   291→  const startSession = () => {
   292→    if (due.length === 0) return;
   293→    finishedRef.current = false;
   294→    setQueue(due);
   295→    setPos(0);
   296→    setFlipped(false);
   297→    setReviewed(0);
   298→  };
   299→
   300→  const endSession = () => {
   301→    setQueue(null);
   302→    setFlipped(false);
   303→  };
   304→
   305→  const review = (correct: boolean) => {
   306→    if (!queue || pos >= queue.length) return;
   307→    const card = queue[pos];
   308→    onCardReview(card.id, correct);
   309→    setFlipped(false);
   310→    setReviewed((r) => r + 1);
   311→    const np = pos + 1;
   312→    setPos(np);
   313→    if (np >= queue.length && !finishedRef.current) {
   314→      finishedRef.current = true;
   315→      onPracticeDone('srs');
   316→    }
   317→  };
   318→
   319→  // Overview
   320→  if (queue === null) {
   321→    return (
   322→      <TabCard>
   323→        <h2 className="font-heading text-lg font-semibold">Flashcards</h2>
   324→        <p className="mt-1 text-sm text-muted-foreground">
   325→          {due.length === 1 ? '1 card due today' : `${due.length} cards due today`}
   326→        </p>
   327→        <p className="mt-1 text-xs text-muted-foreground">
   328→          Box 1: {boxCounts[0]} &middot; Box 2: {boxCounts[1]} &middot; Box 3: {boxCounts[2]} &middot;
   329→          Box 4: {boxCounts[3]} &middot; Box 5: {boxCounts[4]} &middot; Not started: {unseen}
   330→        </p>
   331→
   332→        {due.length === 0 ? (
   333→          <div className="mt-2">
   334→            <EmptyState note="Nothing due right now — cards return after 1, 2, 4, 8 or 16 days depending on the box they reach." />
   335→            {nextDue !== null && (
   336→              <p className="text-center text-sm text-muted-foreground">
   337→                Next cards return on {nextDue}.
   338→              </p>
   339→            )}
   340→          </div>
   341→        ) : (
   342→          <div className="mt-4">
   343→            <Button className="min-h-[44px]" onClick={startSession}>
   344→              Start review session
   345→            </Button>
   346→          </div>
   347→        )}
   348→
   349→        <p className="mt-4 text-xs text-muted-foreground">
   350→          New cards appear as you study chapters.
   351→        </p>
   352→      </TabCard>
   353→    );
   354→  }
   355→
   356→  // Session summary
   357→  if (pos >= queue.length) {
   358→    return (
   359→      <TabCard>
   360→        <div className="flex flex-col items-center gap-3 py-4 text-center">
   361→          <Mascot pose="cheering" size="xl" bob />
   362→          <h2 className="font-heading text-xl font-bold">
   363→            Reviewed {reviewed} card{reviewed === 1 ? '' : 's'}
   364→          </h2>
   365→          <p className="text-sm text-muted-foreground">
   366→            +{PRACTICE_XP.srs} XP{' '}
   367→            {collected ? 'collected today' : "for today's first review set"}
   368→          </p>
   369→          <Button variant="outline" className="min-h-[44px]" onClick={endSession}>
   370→            Back to overview
   371→          </Button>
   372→        </div>
   373→      </TabCard>
   374→    );
   375→  }
   376→
   377→  // One card at a time
   378→  const card = queue[pos];
   379→  return (
   380→    <TabCard>
   381→      <div className="flex items-center justify-between gap-3">
   382→        <h2 className="font-heading text-lg font-semibold">
   383→          Card {pos + 1} of {queue.length}
   384→        </h2>
   385→        <Button variant="ghost" className="min-h-[44px]" onClick={endSession}>
   386→          End session
   387→        </Button>
   388→      </div>
   389→      <Progress
   390→        value={(pos / queue.length) * 100}
   391→        className="mt-2 h-2"
   392→        aria-label={`Review progress: card ${pos + 1} of ${queue.length}`}
   393→      />
   394→
   395→      <div className="mt-4 rounded-2xl border border-border bg-background p-5 text-center">
   396→        <Badge variant="secondary">{card.kind === 'qa' ? 'Question' : 'Term'}</Badge>
   397→        <p className="mt-3 font-heading text-lg font-semibold sm:text-xl">{card.front}</p>
   398→        {flipped && (
   399→          <div
   400→            role="status"
   401→            aria-live="polite"
   402→            className="anim-pop mt-4 rounded-xl bg-success-soft p-4 text-success-soft-foreground"
   403→          >
   404→            <p className="text-sm font-bold">Answer</p>
   405→            <p className="mt-1 text-sm">{card.back}</p>
   406→          </div>
   407→        )}
   408→      </div>
   409→
   410→      <div className="mt-4 flex flex-wrap items-center justify-center gap-2">
   411→        {!flipped ? (
   412→          <Button className="min-h-[44px]" onClick={() => setFlipped(true)}>
   413→            <Eye className="size-4" aria-hidden="true" /> Flip card
   414→          </Button>
   415→        ) : (
   416→          <>
   417→            <Button
   418→              className="min-h-[44px] bg-success text-success-foreground hover:bg-success/90"
   419→              onClick={() => review(true)}
   420→            >
   421→              <Check className="size-4" aria-hidden="true" /> I knew it
   422→            </Button>
   423→            <Button variant="outline" className="min-h-[44px]" onClick={() => review(false)}>
   424→              See it again
   425→            </Button>
   426→          </>
   427→        )}
   428→      </div>
   429→    </TabCard>
   430→  );
   431→}

   483→// ---------- 3. Timed drill ----------
   484→
   485→function DrillTab({
   486→  state,
   487→  today,
   488→  band,
   489→  onPracticeDone,
   490→}: {
   491→  state: LumioState;
   492→  today: string;
   493→  band: Band;
   494→  onPracticeDone: (kind: PracticeKind) => void;
   495→}) {
   496→  const pool = BANDS[band].chapters.flatMap((ch, ci) =>
   497→    ch.questions.map((q, qi) => ({ key: `${band}#${ci}#${qi}`, q, chapterTitle: ch.title })),
   498→  );
   499→
   500→  const [phase, setPhase] = useState<'idle' | 'run' | 'done'>('idle');
   501→  const [untimed, setUntimed] = useState(false);
   502→  const [order, setOrder] = useState<number[] | null>(null);
   503→  const [pos, setPos] = useState(0);
   504→  const [score, setScore] = useState(0);
   505→  const [answered, setAnswered] = useState(0);
   506→  const [picked, setPicked] = useState<number | null>(null);
   507→  const [secondsLeft, setSecondsLeft] = useState<number | null>(null);
   508→  const [deadline, setDeadline] = useState<number | null>(null);
   509→  const [announce, setAnnounce] = useState('');
   510→
   511→  const endedRef = useRef(false);
   512→  const announcedRef = useRef<Set<number>>(new Set());
   513→  const timeoutRef = useRef<number | null>(null);
   514→  const endRoundRef = useRef<() => void>(() => {});
   515→
   516→  // Derived values (declared above all effects, per project rule).
   517→  const running = phase === 'run' && order !== null;
   518→  const cur = order !== null ? pool[order[pos] % pool.length] : null;
   519→  const collected = practiceDoneToday(state, 'drill', today);
   520→
   521→  const endRound = () => {
   522→    if (endedRef.current) return;
   523→    endedRef.current = true;
   524→    if (timeoutRef.current !== null) {
   525→      window.clearTimeout(timeoutRef.current);
   526→      timeoutRef.current = null;
   527→    }
   528→    setPicked(null);
   529→    setPhase('done');
   530→    onPracticeDone('drill');
   531→  };
   532→  endRoundRef.current = endRound;
   533→
   534→  const finishAnswer = () => {
   535→    if (timeoutRef.current !== null) {
   536→      window.clearTimeout(timeoutRef.current);
   537→      timeoutRef.current = null;
   538→    }
   539→    if (endedRef.current) return;
   540→    const na = answered + 1;
   541→    if (untimed && na >= DRILL_UNTIMED_COUNT) {
   542→      setAnswered(na);
   543→      endRound();
   544→      return;
   545→    }
   546→    setAnswered(na);
   547→    setPicked(null);
   548→    setPos((p) => (p + 1) % Math.max(1, order?.length ?? 1));
   549→  };
   550→
   551→  const pick = (i: number) => {
   552→    if (!running || picked !== null || !cur) return;
   553→    const correct = i === cur.q.answer;
   554→    setPicked(i);
   555→    if (correct) {
   556→      setScore((s) => s + 1);
   557→      finishAnswer();
   558→    } else {
   559→      // Wrong: show the explanation briefly, then move on.
   560→      timeoutRef.current = window.setTimeout(finishAnswer, 1600);
   561→    }
   562→  };
   563→
   564→  const startRound = () => {
   565→    endedRef.current = false;
   566→    announcedRef.current = new Set();
   567→    const seed = (Date.now() % 2147483647) || 1;
   568→    setOrder(seededShuffle(pool.map((_, i) => i), seed));
   569→    setPos(0);
   570→    setScore(0);
   571→    setAnswered(0);
   572→    setPicked(null);
   573→    setSecondsLeft(untimed ? null : Math.ceil(DRILL_MS / 1000));
   574→    setDeadline(untimed ? null : Date.now() + DRILL_MS);
   575→    setAnnounce(untimed ? 'Untimed round started. Ten questions.' : 'Round started. Sixty seconds.');
   576→    setPhase('run');
   577→  };
   578→
   579→  // Countdown: ticks every 250ms but re-renders only when the second changes.
   580→  useEffect(() => {
   581→    if (!running || deadline === null) return;
   582→    const t = window.setInterval(() => {
   583→      const s = Math.ceil((deadline - Date.now()) / 1000);
   584→      setSecondsLeft((prev) => (prev === s ? prev : s));
   585→      if (s <= 0) {
   586→        window.clearInterval(t);
   587→        endRoundRef.current();
   588→      }
   589→    }, 250);
   590→    return () => window.clearInterval(t);
   591→  }, [running, deadline]);
   592→
   593→  // Announce time politely only at 30s and 10s — never every second.
   594→  useEffect(() => {
   595→    if (!running || secondsLeft === null) return;
   596→    if ((secondsLeft === 30 || secondsLeft === 10) && !announcedRef.current.has(secondsLeft)) {
   597→      announcedRef.current.add(secondsLeft);
   598→      setAnnounce(`${secondsLeft} seconds left.`);
   599→    }
   600→  }, [running, secondsLeft]);
   601→
   602→  useEffect(
   603→    () => () => {
   604→      if (timeoutRef.current !== null) window.clearTimeout(timeoutRef.current);
   605→    },
   606→    [],
   607→  );
   608→
   609→  if (phase === 'idle') {
   610→    return (
   611→      <TabCard>
   612→        <h2 className="font-heading text-lg font-semibold">Timed drill</h2>
   613→        <p className="mt-1 text-sm text-muted-foreground">
   614→          A 60-second rapid-fire round across every chapter in this band. A wrong answer shows the
   615→          explanation for a moment, then moves on. The pool loops if you outrun it.
   616→        </p>
   617→        <div className="mt-4 flex flex-wrap items-center gap-3">
   618→          <Button className="min-h-[44px]" onClick={startRound}>
   619→            <Timer className="size-4" aria-hidden="true" /> Start drill
   620→          </Button>
   621→          <button
   622→            type="button"
   623→            aria-pressed={untimed}
   624→            onClick={() => setUntimed(!untimed)}
   625→            className={cn(
   626→              'press flex min-h-[44px] items-center gap-2 rounded-full border px-4 text-sm font-medium',
   627→              untimed ? 'border-primary bg-primary/10 text-primary' : 'border-border',
   628→            )}
   629→          >
   630→            <Clock className="size-4" aria-hidden="true" />
   631→            Untimed round {untimed ? 'on' : 'off'}
   632→          </button>
   633→        </div>
   634→        <p className="mt-3 text-xs text-muted-foreground">
   635→          Untimed rounds run 10 questions with no clock.
   636→        </p>
   637→      </TabCard>
   638→    );
   639→  }
   640→
   641→  if (phase === 'done') {
   642→    return (
   643→      <TabCard>
   644→        <div className="flex flex-col items-center gap-3 py-4 text-center">
   645→          <Mascot pose="excited" size="xl" bob />
   646→          <h2 className="font-heading text-xl font-bold">{score} correct</h2>
   647→          <p className="text-sm text-muted-foreground">
   648→            {answered} question{answered === 1 ? '' : 's'} answered this round.
   649→          </p>
   650→          <p className="text-sm text-muted-foreground">
   651→            +{PRACTICE_XP.drill} XP {collected ? 'collected today' : "for today's first round"}
   652→          </p>
   653→          <Button className="min-h-[44px]" onClick={() => setPhase('idle')}>
   654→            New round
   655→          </Button>
   656→        </div>
   657→      </TabCard>
   658→    );
   659→  }
   660→
   661→  // Running
   662→  if (!cur) return null;
   663→  return (
   664→    <TabCard>
   665→      <div className="flex flex-wrap items-center justify-between gap-2">
   666→        {secondsLeft !== null ? (
   667→          <Badge variant="secondary" className="font-mono text-base">
   668→            {secondsLeft}s left
   669→          </Badge>
   670→        ) : (
   671→          <Badge variant="secondary">Untimed</Badge>
   672→        )}
   673→        <p className="text-sm text-muted-foreground">
   674→          Score {score} &middot;{' '}
   675→          {untimed
   676→            ? `${Math.min(answered + 1, DRILL_UNTIMED_COUNT)} of ${DRILL_UNTIMED_COUNT}`
   677→            : `${answered} answered`}
   678→        </p>
   679→      </div>
   680→      {untimed && (
   681→        <Progress
   682→          value={(answered / DRILL_UNTIMED_COUNT) * 100}
   683→          className="mt-2 h-2"
   684→          aria-label={`Drill progress: ${answered} of ${DRILL_UNTIMED_COUNT} questions answered`}
   685→        />
   686→      )}
   687→      <p aria-live="polite" className="mt-1 min-h-[1rem] text-xs text-muted-foreground">
   688→        {announce}
   689→      </p>
   690→
   691→      <p className="mt-2 font-heading text-base font-semibold sm:text-lg">{cur.q.text}</p>
   692→      <p className="mt-1 text-xs text-muted-foreground">{cur.chapterTitle}</p>
   693→
   694→      <div className="mt-3 flex flex-col gap-2">
   695→        {cur.q.options.map((opt, i) => (
   696→          <OptionButton
   697→            key={i}
   698→            label={opt}
   699→            index={i}
   700→            state={
   701→              picked === null
   702→                ? 'idle'
   703→                : picked === i
   704→                  ? i === cur.q.answer
   705→                    ? 'correct'
   706→                    : 'wrong'
   707→                  : 'dim'
   708→            }
   709→            onClick={() => pick(i)}
   710→          />
   711→        ))}
   712→      </div>
   713→
   714→      {picked !== null && picked !== cur.q.answer && (
   715→        <div
   716→          role="status"
   717→          aria-live="polite"
   718→          className="anim-pop mt-3 rounded-xl bg-gold-soft p-4"

   725→          <p className="mt-1 text-sm">{cur.q.explanation}</p>

   848→        value={(pos / ADAPTIVE_COUNT) * 100}
   849→        className="mt-2 h-2"

  1032→function RevisionTab({
  1033→  state,
  1034→  today,
  1035→  band,
  1036→  onPracticeDone,
  1037→}: {
  1038→  state: LumioState;
  1039→  today: string;
  1040→  band: Band;
  1041→  onPracticeDone: (kind: PracticeKind) => void;
  1042→}) {
  1043→  const chapters = BANDS[band].chapters;
  1044→  const completed = chapters
  1045→    .map((_, i) => i)
  1046→    .filter((i) => (state.res[chapterId(band, i)]?.best ?? 0) >= 1);
  1047→  const locked = completed.length < 2;
  1048→  const collected = practiceDoneToday(state, 'revision', today);
  1049→
  1050→  const [round, setRound] = useState(0);
  1051→
  1052→  const items: QuizItem[] = [];
  1053→  if (!locked) {
  1054→    const perCh = Math.ceil(REVISION_COUNT / completed.length);
  1055→    const queues = completed.map((ci, k) => {
  1056→      const qs = chapters[ci].questions;
  1057→      const expanded: Array<{ q: Question; qi: number }> = [];
  1058→      for (let n = 0; expanded.length < perCh; n++) {
  1059→        const qi = n % qs.length;
  1060→        expanded.push({ q: qs[qi], qi });
  1061→      }
  1062→      return seededShuffle(expanded, (round + 1) * 97 + k * 13 + 7).map(({ q, qi }) => ({
  1063→        key: `${band}#${ci}#${qi}`,
  1064→        q,
  1065→        chapterTitle: chapters[ci].title,
  1066→      }));
  1067→    });
  1068→    for (let r = 0; items.length < REVISION_COUNT; r++) {
  1069→      let any = false;
  1070→      for (const queue of queues) {
  1071→        if (items.length >= REVISION_COUNT) break;
  1072→        const it = queue[r];
  1073→        if (it) {
  1074→          items.push(it);
  1075→          any = true;
  1076→        }
  1077→      }
  1078→      if (!any) break;
  1079→    }
  1080→  }
  1081→
  1082→  if (locked) {
  1083→    return (
  1084→      <TabCard>
  1085→        <h2 className="font-heading text-lg font-semibold">Mixed revision</h2>
  1086→        <EmptyState note="Complete at least two chapter quizzes first — revision mixes what you have already learned." />
  1087→        <p className="text-center text-sm text-muted-foreground">
  1088→          Chapters completed so far: {completed.length}
  1089→        </p>
  1090→      </TabCard>
  1091→    );
  1092→  }
  1093→
  1094→  return (
  1095→    <div className="flex flex-col gap-3">
  1096→      <TabCard>
  1097→        <h2 className="font-heading text-lg font-semibold">Mixed revision</h2>
  1098→        <p className="mt-1 text-sm text-muted-foreground">
  1099→          Ten questions interleaved across your {completed.length} completed chapter
  1100→          {completed.length === 1 ? '' : 's'} — revising in mixed order is harder and it sticks
  1101→          better.
  1102→        </p>
  1103→      </TabCard>
  1104→      <RunQuiz
  1105→        key={round}
  1106→        items={items}
  1107→        doneTitle="Mixed revision complete."
  1108→        doneXpLine={`+${PRACTICE_XP.revision} XP ${
  1109→          collected ? 'collected today' : "for today's first revision"
  1110→        }`}
  1111→        onDone={() => onPracticeDone('revision')}
  1112→        onAgain={() => setRound((r) => r + 1)}
  1113→      />
  1114→    </div>
  1115→  );
  1116→}

  1118→// ---------- 6. Legendary ----------

  1120→function LegendaryTab({
  1121→  state,
  1122→  band,
  1123→  onLegendary,
  1124→}: {
  1125→  state: LumioState;
  1126→  band: Band;
  1127→  onLegendary: () => void;
  1128→}) {
  1129→  const chapters = BANDS[band].chapters;
  1130→  const mastered = masteredCount(state);
  1131→  const locked = mastered < 2;
  1132→  const doneEver = legendaryDone(state);
  1133→
  1134→  const [round, setRound] = useState(0);
  1135→
  1136→  // Hard mix: the last question of each chapter, padded with earlier ones if needed.
  1137→  const items: QuizItem[] = [];
  1138→  if (!locked) {
  1139→    for (let back = 0; items.length < LEGENDARY_COUNT && back < 64; back++) {
  1140→      for (let ci = 0; ci < chapters.length && items.length < LEGENDARY_COUNT; ci++) {
  1141→        const qs = chapters[ci].questions;
  1142→        const qi = qs.length - 1 - back;
  1143→        if (qi >= 0) {

  1195→function ErrorLabTab({
  1196→  state,
  1197→  today,
  1198→  band,
  1199→  onPracticeDone,
  1200→}: {
  1201→  state: LumioState;
  1202→  today: string;
  1203→  band: Band;
  1204→  onPracticeDone: (kind: PracticeKind) => void;
  1205→}) {
  1206→  const missed = resolveMissed(band, state.wrong);
  1207→  const collected = practiceDoneToday(state, 'errorlab', today);
  1208→
  1209→  const [pos, setPos] = useState(0);
  1210→  const [done, setDone] = useState(false);
  1211→  const paidRef = useRef(false);
  1212→
  1213→  const advance = () => {
  1214→    if (pos === 0 && !paidRef.current) {
  1215→      paidRef.current = true;
  1216→      onPracticeDone('errorlab');
  1217→    }
  1218→    if (pos + 1 >= missed.length) {
  1219→      setDone(true);
  1220→    } else {
  1221→      setPos(pos + 1);
  1222→    }
  1223→  };
  1224→
  1225→  if (missed.length === 0) {
  1226→    return (
  1227→      <TabCard>
  1228→        <h2 className="font-heading text-lg font-semibold">Error lab</h2>
  1229→        <EmptyState note="No misses recorded yet — the Error Lab fills up as questions come back wrong in quizzes and practice." />
  1230→      </TabCard>
  1231→    );
  1232→  }
  1233→
  1234→  if (done) {
  1235→    return (
  1236→      <TabCard>
  1237→        <div className="flex flex-col items-center gap-3 py-4 text-center">
  1238→          <Mascot pose="excited" size="xl" bob />
  1239→          <h2 className="font-heading text-xl font-bold">
  1240→            Reviewed {missed.length} mistake{missed.length === 1 ? '' : 's'}
  1241→          </h2>
  1242→          <p className="text-sm text-muted-foreground">
  1243→            +{PRACTICE_XP.errorlab} XP {collected ? 'collected today' : "for today's first lab"}
  1244→          </p>
  1245→          <Button
  1246→            variant="outline"
  1247→            className="min-h-[44px]"
  1248→            onClick={() => {
  1249→              setPos(0);
  1250→              setDone(false);
  1251→              paidRef.current = true;
  1252→            }}
  1253→          >
  1254→            Review them again
  1255→          </Button>
  1256→        </div>
  1257→      </TabCard>
  1258→    );
  1259→  }
  1260→
  1261→  const row = missed[pos];
  1262→  const tagged = state.mistakes[row.id]?.tag;
  1263→  const tagLabel = MISTAKE_TAGS.find((t) => t.tag === tagged)?.label;
  1264→  const steps = workedSteps(row.q, row.chapterTitle, row.subject);
  1265→  const last = pos + 1 >= missed.length;
  1266→
  1267→  return (
  1268→    <TabCard>
  1269→      <div className="flex items-center justify-between gap-3">
  1270→        <h2 className="font-heading text-lg font-semibold">
  1271→          Mistake {pos + 1} of {missed.length}
  1272→        </h2>
  1273→        <Badge variant="secondary">{row.subject}</Badge>
  1274→      </div>
  1275→
  1276→      <p className="mt-3 font-heading text-base font-semibold sm:text-lg">{row.q.text}</p>
  1277→      <p className="mt-1 text-xs text-muted-foreground">
  1278→        You missed this in &ldquo;{row.chapterTitle}&rdquo;
  1279→        {tagLabel ? ` — tagged as ${tagLabel.toLowerCase()}` : ''}.
  1280→      </p>
  1281→
  1282→      <div className="mt-4 rounded-2xl border border-border bg-background p-4">
  1283→        <p className="text-sm font-semibold">Worked solution</p>
  1284→        <ol className="mt-3 flex flex-col gap-3">
  1285→          {steps.map((step, i) => (
  1286→            <li key={step.title} className="flex items-start gap-3">
  1287→              <span
  1288→                className="flex size-7 shrink-0 items-center justify-center rounded-full bg-primary/10 text-xs font-bold text-primary"
  1289→                aria-hidden="true"
  1290→              >
  1291→                {i + 1}
  1292→              </span>
  1293→              <div>
  1294→                <p className="text-sm font-semibold">{step.title}</p>
  1295→                <p className="mt-0.5 text-sm text-muted-foreground">{step.body}</p>
  1296→              </div>
  1297→            </li>
  1298→          ))}
  1299→        </ol>
  1300→      </div>
  1301→
  1302→      <div className="mt-5 flex items-center justify-end">
  1303→        <Button className="min-h-[44px]" onClick={advance}>
  1304→          {last ? 'Finish review' : 'Next mistake'} <ArrowRight className="size-4" aria-hidden="true" />
  1305→        </Button>
  1306→      </div>
  1307→    </TabCard>
  1308→  );
  1309→}
