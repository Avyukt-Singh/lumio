    17→    17→export function dayKey(d: Date, tz: string): string {
    18→    18→  const fmt = new Intl.DateTimeFormat('en-CA', {
    19→    19→    timeZone: tz,
    20→    20→    year: 'numeric',
    21→    21→    month: '2-digit',
    22→    22→    day: '2-digit',
    23→    23→  });
    24→    24→  return fmt.format(d); // en-CA gives yyyy-mm-dd
    25→    25→}

    33→    33→export function weekdayOf(d: Date, tz: string): number {
    34→    34→  // 0 = Sunday ... 6 = Saturday, computed from the learner-local date
    35→    35→  const wd = new Intl.DateTimeFormat('en-US', { timeZone: tz, weekday: 'short' }).format(d);

    39→    39→/** Monday-start ISO week key like 2025-W45 (5.6: weeks start on Monday). */
    40→    40→export function weekKey(d: Date, tz: string): string {
    41→    41→  const dateStr = dayKey(d, tz);
    42→    42→  const dt = new Date(dateStr + 'T12:00:00Z');
    43→    43→  const day = (dt.getUTCDay() + 6) % 7; // Mon=0
    44→    44→  dt.setUTCDate(dt.getUTCDate() - day + 3); // shift to Thursday of this ISO week
    45→    45→  const isoYear = dt.getUTCFullYear();
    46→    46→  const jan4 = new Date(Date.UTC(isoYear, 0, 4));
    47→    47→  const jan4Day = (jan4.getUTCDay() + 6) % 7;
    48→    48→  const firstThu = new Date(jan4);
    49→    49→  firstThu.setUTCDate(jan4.getUTCDate() - jan4Day + 3);
    50→    50→  const weekNum = 1 + Math.round((dt.getTime() - firstThu.getTime()) / (7 * 86400000));
    51→    51→  return `${isoYear}-W${String(Math.max(1, weekNum)).padStart(2, '0')}`;
    52→    52→}

    98→    98→ * Streak = consecutive learner-local days with an XP-earning activity, counting back
    99→    99→ * from today, or from yesterday if today is not yet done. Opening the app never counts.
   100→   100→ * A spent streak freeze (`freezeUsed[date]`) shields one missed day, keeping the chain
   101→   101→ * alive (chunk 4A, 7.10).

   103→   103→export function streakFrom(
   104→   104→  days: string[],
   105→   105→  today: string,
   106→   106→  freezeUsed?: Record<string, true>
   107→   107→): number {
   108→   108→  const set = new Set(days);
   109→   109→  const shield = freezeUsed ?? {};
   110→   110→  const active = (d: string) => set.has(d) || shield[d] === true;
   111→   111→  let cursor = active(today) ? today : addDays(today, -1);
   112→   112→  let n = 0;
   113→   113→  while (active(cursor)) {
   114→   114→    n += 1;
   115→   115→    cursor = addDays(cursor, -1);
   116→   116→  }
   117→   117→  return n;
   118→   118→}

   140→   140→export function chapterState(
   141→   141→  band: Band,
   142→   142→  index: number,
   143→   143→  res: Record<string, ChapterResult>
   144→   144→): ChapterState {
   145→   145→  const nq = BANDS[band].chapters[index]?.questions.length ?? 0;
   146→   146→  const best = bestScore(res, chapterId(band, index));

   150→   150→  return bestScore(res, chapterId(band, index - 1)) >= 1 ? 'available' : 'locked';
   151→   151→}

   166→   166→// ---------- scheduler (3.6 / 7.3 basics) ----------
   167→   167→// Sessions are placed after school, around dinner, before bedtime, with 15-minute gaps.
   168→   168→// Busy windows: school hours and dinner (45 minutes).
   169→   169→
   170→   170→export interface SessionSlot {
   171→   171→  startMin: number;
   172→   172→  endMin: number;
   173→   173→  label: string;
   174→   174→}
   175→   175→
   176→   176→export const GAP_MIN = 15;
   177→   177→export const DINNER_LEN = 45;
   178→   178→
   179→   179→/** Free study windows for a day: after school/work and after dinner, minus commitment blocks. */
   180→   180→function freeWindows(plan: SchedulePlan): Array<[number, number]> {
   181→   181→  const { startMin: _s, endMin, dinnerMin, bedMin, sessionLen } = plan;
   182→   182→  let free: Array<[number, number]> = [];
   183→   183→  if (endMin + GAP_MIN <= dinnerMin) free.push([endMin + GAP_MIN, dinnerMin]);
   184→   184→  if (dinnerMin + DINNER_LEN <= bedMin) free.push([dinnerMin + DINNER_LEN, bedMin]);
   185→   185→
   186→   186→  // subtract commitment blocks (clamped to sensible bounds) from the free windows
   187→   187→  const blocks = (plan.busy ?? [])
   188→   188→    .filter((c) => c.endMin > c.startMin)
   189→   189→    .map((c) => [Math.max(0, c.startMin), Math.min(24 * 60, c.endMin)] as [number, number])
   190→   190→    .sort((a, b) => a[0] - b[0]);
   191→   191→  for (const [bs, be] of blocks) {
   192→   192→    const next: Array<[number, number]> = [];
   193→   193→    for (const [ws, we] of free) {
   194→   194→      if (be <= ws || bs >= we) {
   195→   195→        next.push([ws, we]); // no overlap
   196→   196→        continue;
   197→   197→      }
   198→   198→      if (bs > ws) next.push([ws, Math.min(bs, we)]);
   199→   199→      if (be < we) next.push([Math.max(be, ws), we]);
   200→   200→    }
   201→   201→    free = next.filter(([ws, we]) => we - ws >= Math.min(10, sessionLen));
   202→   202→  }
   203→   203→  return free;
   204→   204→}
   205→   205→
   206→   206→/**
   207→   207→ * Lay out `plan.sessions` study sessions for one day. Sessions go after
   208→   208→ * school/work and after dinner, before bedtime, with 15-minute gaps. Structured
   209→   209→ * commitment blocks (7.1 upgrade) are subtracted from the free windows —
   210→   210→ * the planner never overlaps a commitment. Returns the slots, or a
   211→   211→ * "day too full" verdict instead of guessing (3.3: if the day is too full say so).
   212→   212→ */
   213→   213→export function planDay(plan: SchedulePlan): { slots: SessionSlot[]; dropped: number } {
   214→   214→  const { sessions, sessionLen } = plan;
   215→   215→  const free = freeWindows(plan);
   216→   216→
   217→   217→  const slots: SessionSlot[] = [];
   218→   218→  for (const [ws, we] of free) {
   219→   219→    let t = ws;
   220→   220→    while (slots.length < sessions && t + sessionLen <= we) {
   221→   221→      slots.push({
   222→   222→        startMin: t,
   223→   223→        endMin: t + sessionLen,
   224→   224→        label: sessions === 1 ? 'Study session' : `Session ${slots.length + 1}`,
   225→   225→      });
   226→   226→      t += sessionLen + GAP_MIN;
   227→   227→    }
   228→   228→    if (slots.length >= sessions) break;
   229→   229→  }
   230→   230→  return { slots, dropped: sessions - slots.length };
   231→   231→}
   232→   232→
   233→   233→/**
   234→   234→ * The day's plan with the learner's interactive reschedule overrides applied
   235→   235→ * (chunk 4A). Overrides are validated when generated (rescheduleOptions); an
   236→   236→ * override for a slot index that does not exist is ignored.
   237→   237→ */
   238→   238→export function planDayWithOverrides(
   239→   239→  plan: SchedulePlan,
   240→   240→  overrides?: Record<string, number>
   241→   241→): { slots: SessionSlot[]; dropped: number } {
   242→   242→  const base = planDay(plan);
   243→   243→  if (!overrides || Object.keys(overrides).length === 0) return base;
   244→   244→  const slots = base.slots.map((s, i) => {
   245→   245→    const ov = overrides[String(i)];
   246→   246→    if (typeof ov !== 'number') return s;
   247→   247→    return { ...s, startMin: ov, endMin: ov + (plan.sessionLen ?? s.endMin - s.startMin) };
   248→   248→  });
   249→   249→  return { slots, dropped: base.dropped };
   250→   250→}
   251→   251→
   252→   252→/**
   253→   253→ * Every valid 15-minute-aligned start time for moving session `index`: inside a
   254→   254→ * free window (school/dinner/commitments honoured), not overlapping another
   255→   255→ * placed session, and not its own current start.
   256→   256→ */
   257→   257→export function rescheduleOptions(
   258→   258→  plan: SchedulePlan,
   259→   259→  overrides: Record<string, number> | undefined,
   260→   260→  index: number
   261→   261→): number[] {
   262→   262→  const { slots } = planDay(plan);
   263→   263→  if (index < 0 || index >= slots.length) return [];
   264→   264→  const len = plan.sessionLen;
   265→   265→  const others = slots
   266→   266→    .map((s, i) => {
   267→   267→      const ov = overrides?.[String(i)];
   268→   268→      const start = typeof ov === 'number' ? ov : s.startMin;
   269→   269→      return { i, startMin: start, endMin: start + len };
   270→   270→    })
   271→   271→    .filter((s) => s.i !== index);
   272→   272→  const out: number[] = [];
   273→   273→  for (const [ws, we] of freeWindows(plan)) {
   274→   274→    for (let t = ws; t + len <= we; t += 15) {
   275→   275→      const clash = others.some((o) => t < o.endMin && o.startMin < t + len);
   276→   276→      if (!clash && t !== slots[index].startMin) out.push(t);
   277→   277→    }
   278→   278→  }
   279→   279→  return out;
   280→   280→}

   282→   282→/**
   283→   283→ * Subject pairing (7.3): even-indexed sessions carry a core subject that rotates
   284→   284→ * day by day; odd-indexed sessions pair with the complementary subject when one
   285→   285→ * is set (session 2 pairs with the same core as session 1, keeping the pair). */
   286→   286→export function subjectForSession(
   287→   287→  subjects: string[],
   288→   288→  compSub: string | undefined,
   289→   289→  dayIndex: number,
   290→   290→  sessionIndex: number
   291→   291→): string | null {
   292→   292→  if (subjects.length === 0 && !compSub) return null;
   293→   293→  if (sessionIndex % 2 === 1 && compSub) return compSub;
   294→   294→  const pool = subjects.length > 0 ? subjects : compSub ? [compSub] : [];
   295→   295→  if (pool.length === 0) return null;
   296→   296→  const pairIndex = Math.floor(sessionIndex / 2);
   297→   297→  return pool[(dayIndex + pairIndex) % pool.length];
   298→   298→}

   404→   404→export function questProgress(
   405→   405→  state: Pick<

   408→   408→  >,

   413→   413→  const wk = weekKey(now, tz);
   414→   414→  const mk = monthKey(now, tz);
   415→   415→  const weekDays = new Set(weekDatesOf(today));
   416→   416→  return QUESTS.map((q) => {

   432→   432→          .filter((e) => e.source === 'practice' && weekDays.has(e.date))
   433→   433→          .map((e) => e.date)
   434→   434→      );

   600→   600→export interface WeakTopic {
   601→   601→  chapterIndex: number;
   602→   602→  chapterTitle: string;
   603→   603→  subject: string;
   604→   604→  /** distinct missed question ids in this chapter */
   605→   605→  misses: number;
   606→   606→  /** chapter weight emphasises earlier fundamentals (2/3/1/2/1/2 ranking heuristic) */
   607→   607→  weight: number;
   608→   608→}
   609→   609→
   610→   610→/**
   611→   611→ * Weak-topic ranking from `state.wrong` (stable ids `${band}#${ch}#${q}`),
   612→   612→ * merged with the chapters the learner marked weak in self-reported mocks
   613→   613→ * (chunk 4A — each mock marking adds one miss-like weight). Chapters are
   614→   614→ * ranked by distinct misses, then by an importance weight that emphasises
   615→   615→ * the chapters the study path treats as foundations.
   616→   616→ */
   617→   617→export function weakTopics(state: LumioState, limit = 6): WeakTopic[] {
   618→   618→  if (!state.profile) return [];
   619→   619→  const band = state.profile.band;
   620→   620→  const counts = new Map<number, Set<string>>();
   621→   621→  for (const id of Object.keys(state.wrong)) {
   622→   622→    const [b, ch] = id.split('#');
   623→   623→    if (b !== band) continue;
   624→   624→    const idx = Number(ch);
   625→   625→    if (!Number.isFinite(idx)) continue;
   626→   626→    if (!counts.has(idx)) counts.set(idx, new Set());
   627→   627→    counts.get(idx)!.add(id);
   628→   628→  }
   629→   629→  // mock-reported weak chapters merge into the same ranking
   630→   630→  for (const mock of state.mocks ?? []) {
   631→   631→    for (const idx of mock.weakChapters ?? []) {
   632→   632→      if (!counts.has(idx)) counts.set(idx, new Set());
   633→   633→      counts.get(idx)!.add(`mock#${mock.date}#${mock.subject}`);
   634→   634→    }
   635→   635→  }
   636→   636→  const chapters = BANDS[band].chapters;
   637→   637→  const weightFor = (i: number) => [2, 3, 1, 2, 1, 2][i % 6];
   638→   638→  const rows: WeakTopic[] = [];
   639→   639→  for (const [idx, ids] of counts) {
   640→   640→    const ch = chapters[idx];
   641→   641→    if (!ch) continue;
   642→   642→    rows.push({
   643→   643→      chapterIndex: idx,
   644→   644→      chapterTitle: ch.title,
   645→   645→      subject: ch.subject,
   646→   646→      misses: ids.size,
   647→   647→      weight: weightFor(idx),
   648→   648→    });
   649→   649→  }
   650→   650→  rows.sort((a, b) => b.misses * b.weight - a.misses * a.weight || a.chapterIndex - b.chapterIndex);
   651→   651→  return rows.slice(0, limit);
   652→   652→}

   678→   678→/**
   679→   679→ * A Monday-start month grid (6 rows x 7 cols). Cells outside the month are
   680→   680→ * null, so the calendar never invents neighbouring-month data.
   681→   681→ */
   682→   682→export function monthCells(year: number, month0: number): Array<string | null> {
   683→   683→  const first = new Date(Date.UTC(year, month0, 1));
   684→   684→  const lead = (first.getUTCDay() + 6) % 7; // Mon=0 blank cells
   685→   685→  const dayCount = new Date(Date.UTC(year, month0 + 1, 0)).getUTCDate();
   686→   686→  const cells: Array<string | null> = Array.from({ length: lead }, () => null);
   687→   687→  for (let d = 1; d <= dayCount; d++) {
   688→   688→    cells.push(
   689→   689→      `${year}-${String(month0 + 1).padStart(2, '0')}-${String(d).padStart(2, '0')}`
   690→   690→    );
   691→   691→  }
   692→   692→  while (cells.length % 7 !== 0) cells.push(null);
   693→   693→  return cells;
   694→   694→}

   696→   696→function pad(n: number): string {
   697→   697→  return String(n).padStart(2, '0');

   700→   700→/**
   701→   701→ * Convert a learner-local wall-clock time (date + minutes) in `tz` to the true
   702→   702→ * UTC instant. Iterative offset resolution handles DST boundaries honestly.
   703→   703→ */
   704→   704→export function zonedMinutesToUtc(dateStr: string, minutes: number, tz: string): Date {
   705→   705→  const hh = Math.floor(minutes / 60) % 24;
   706→   706→  const mm = minutes % 60;
   707→   707→  const naive = Date.parse(`${dateStr}T${pad(hh)}:${pad(mm)}:00Z`);
   708→   708→  let ts = naive;
   709→   709→  const fmt = new Intl.DateTimeFormat('en-CA', {
   710→   710→    timeZone: tz,
   711→   711→    year: 'numeric',
   712→   712→    month: '2-digit',
   713→   713→    day: '2-digit',
   714→   714→    hour: '2-digit',
   715→   715→    minute: '2-digit',
   716→   716→    second: '2-digit',
   717→   717→    hour12: false,
   718→   718→  });
   719→   719→  for (let i = 0; i < 3; i++) {
   720→   720→    const parts = fmt.formatToParts(new Date(ts));
   721→   721→    const get = (t: string) => Number(parts.find((p) => p.type === t)?.value ?? 0);
   722→   722→    const asUtc = Date.UTC(
   723→   723→      get('year'),
   724→   724→      get('month') - 1,
   725→   725→      get('day'),
   726→   726→      get('hour') % 24,
   727→   727→      get('minute'),
   728→   728→      get('second')
   729→   729→    );
   730→   730→    ts = naive - (asUtc - naive);
   731→   731→  }
   732→   732→  return new Date(ts);
   733→   733→}

   759→   759→ * without requiring a VTIMEZONE block.

   761→   761→export function buildIcs(events: IcsEvent[], tz: string, calName: string): string {

   772→   772→    const start = zonedMinutesToUtc(ev.date, ev.startMin, tz);

   774→   774→    lines.push(
   775→   775→      'BEGIN:VEVENT',
   776→   776→      `UID:${ev.uid}@lumio.app`,
   777→   777→      `DTSTAMP:${stamp}`,
   778→   778→      `DTSTART:${icsStamp(start)}`,
   779→   779→      `DTEND:${icsStamp(end)}`,
   780→   780→      `SUMMARY:${icsEscape(ev.summary)}`,

   783→   783→    );
   784→   784→  }

   822→   822→/**
   823→   823→ * Validate a study-video URL and extract the 11-character YouTube id.
   824→   824→ * Only youtube.com/watch?v=, youtube.com/shorts/, youtu.be/ and
   825→   825→ * youtube.com/embed/ forms are accepted — every other host is refused.
   826→   826→ */
   827→   827→export function youtubeIdFromUrl(raw: string): string | null {
   828→   828→  const url = raw.trim();
   829→   829→  if (url.length === 0 || url.length > 2048) return null;
   830→   830→  let parsed: URL;
   831→   831→  try {
   832→   832→    parsed = new URL(url);
   833→   833→  } catch {
   834→   834→    return null;
   835→   835→  }
   836→   836→  const host = parsed.hostname.replace(/^www\./i, '').toLowerCase();
   837→   837→  const isYouTube = host === 'youtube.com' || host === 'm.youtube.com' || host === 'youtu.be' || host === 'youtube-nocookie.com';
   838→   838→  if (!isYouTube || parsed.protocol !== 'https:' && parsed.protocol !== 'http:') return null;
   839→   839→  let candidate: string | null = null;
   840→   840→  if (host === 'youtu.be') {
   841→   841→    candidate = parsed.pathname.slice(1).split('/')[0];
   842→   842→  } else if (parsed.pathname === '/watch') {
   843→   843→    candidate = parsed.searchParams.get('v');
   844→   844→  } else {
   845→   845→    const m = parsed.pathname.match(/^\/(?:shorts|embed|live)\/([A-Za-z0-9_-]+)/);
   846→   846→    candidate = m ? m[1] : null;
