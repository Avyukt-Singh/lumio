   851→/** Strict fallback chains (chunk 4B): hi -> hi-Latn -> en; hinglish -> en. */
   852→export function t(lang: UiLang, key: DictKey): string {
   853→  if (lang === 'hi') return HI[key] ?? HINGLISH[key] ?? EN[key];
   854→  if (lang === 'hinglish') return HINGLISH[key] ?? EN[key];
   855→  return EN[key];
   856→}
   857→
   858→/** Convenience: translate for a LumioState profile language (defaults to English). */
   859→export function tFor(lang: UiLang | undefined | null): (key: DictKey) => string {
   860→  return (key: DictKey) => t(lang ?? 'en', key);
   861→}
   862→
   863→export function isUiLang(v: unknown): v is UiLang {
   864→  return v === 'en' || v === 'hi' || v === 'hinglish';
   865→}

   867→/** 'daily-plan-review' -> DictKey 'quest.dailyPlanReview.title' | '.desc' */
   868→function questDictKey(questId: string, kind: 'title' | 'desc'): DictKey {
   869→  const camel = questId
   870→    .split('-')
   871→    .map((part, i) => (i === 0 ? part : part.charAt(0).toUpperCase() + part.slice(1)))
   872→    .join('');
   873→  return `quest.${camel}.${kind}` as DictKey;
   874→}
   875→
   876→export function questTitle(lang: UiLang, questId: string, fallback: string): string {
   877→  const key = questDictKey(questId, 'title');
   878→  return lang === 'en' ? fallback : t(lang, key) || fallback;
   879→}
   880→
   881→export function questDesc(lang: UiLang, questId: string, fallback: string): string {
   882→  const key = questDictKey(questId, 'desc');
   883→  return lang === 'en' ? fallback : t(lang, key) || fallback;
   884→}
