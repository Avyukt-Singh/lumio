    60→/**
    61→ * LumioApp (chunk 2): single-route client app. Owns the typed state (a pure
    62→ * projection of the IndexedDB event log), the theme engine, screen switching
    63→ * and all mutations via `dispatch` — every change is an appended event (ADR
    64→ * 0002). Persistence: event log -> projection -> snapshot mirror.
    65→ */
    66→export function LumioApp() {
    67→  const [state, setState] = useState<LumioState | null>(null);
    68→  const [hydrated, setHydrated] = useState(false);
    69→  const [screen, setScreen] = useState<ScreenId>('home');
    70→  const [reader, setReader] = useState<number | null>(null);
    71→  const [quiz, setQuiz] = useState<number | null>(null);
    72→  const [editProfile, setEditProfile] = useState(false);
    73→  const [saveError, setSaveError] = useState<string | undefined>(undefined);
    74→  const [now, setNow] = useState(() => new Date());
    75→  const [swWaiting, setSwWaiting] = useState<ServiceWorker | null>(null);
    76→  // ---- chunk 5A: account state (never gates the learning loop) ----
    77→  const [accountUser, setAccountUser] = useState<AccountUser | null>(null);
    78→  const [accountAvail, setAccountAvail] = useState<'unknown' | 'ok' | 'unavailable'>('unknown');
    79→  // ---- chunk 5B: AI capability (unknown -> treated as off until the probe resolves) ----
    80→  const [aiMode, setAiMode] = useState<'unknown' | 'off' | 'fake' | 'openai-compatible'>('unknown');
    81→  const [studyBuddyOpen, setStudyBuddyOpen] = useState(false);
    82→  const [devOutbox, setDevOutbox] = useState(false);
    83→  const [devicePrompt, setDevicePrompt] = useState<{ userId: string; userName: string | null } | null>(null);
    84→  const stateRef = useRef<LumioState | null>(null);
    85→  useEffect(() => { stateRef.current = state; }, [state]);
    86→
    87→  // ---- derived values: ALL declared above every useEffect hook ----
    88→  // Plain constants (no useMemo). Keeping every binding above the effects
    89→  // guarantees no hoisted effect can ever read them in their temporal dead zone
    90→  // (the chunk-1 TDZ bug — this ordering is a hard rule for this file).
    91→  const tz = state?.profile?.tz ?? DEFAULT_TZ;
    92→  const calm = state?.settings.calm ?? false;
    93→  const themeOverride = state?.settings.theme ?? 'auto';
    94→  const themeSchedule = state?.settings.sch ?? DEFAULT_SETTINGS.sch;
    95→  const theme: ThemeName =
    96→    themeOverride !== 'auto' ? themeOverride : themeAt(hourOf(now, tz), themeSchedule);
    97→  const band = state?.profile?.band;
    98→  const density =
    99→    band === 'young' ? ('young' as const) : band === 'above' ? ('adult' as const) : ('school' as const);
   100→  // chunk 4B: interface language drives translation + Devanagari typography
   101→  const lang = state?.profile?.lang ?? 'en';
   102→  const tr = tFor(lang);
   103→  // chunk 5B: AI features are hidden entirely until the capability probe says otherwise
   104→  const aiAvailable = aiMode === 'fake' || aiMode === 'openai-compatible';
   105→
   106→  // hydrate once (client only; never during SSR): replay the event log, seeded
   107→  // from the chunk-1 snapshot when present. Falls back to the snapshot mirror
   108→  // if IndexedDB replay fails entirely.
   109→  useEffect(() => {
   110→    let alive = true;
   111→    requestPersistence();
   112→    queueMicrotask(async () => {
   113→      if (!alive) return;
   114→      try {
   115→        const events = await loadEventsAndSeed();
   116→        const projected = replayEvents(events, new Date(), events.length && (events[events.length - 1] as { tz?: string }).tz ? ((events[events.length - 1] as unknown as { tz: string }).tz) : DEFAULT_TZ);
   117→        if (alive) {
   118→          setState(projected);
   119→          saveState(projected);
   120→        }
   121→      } catch {
   122→        const snapshot = loadState();
   123→        if (alive) setState(snapshot ? projectState(normalizeState(snapshot), new Date(), snapshot.profile?.tz ?? DEFAULT_TZ) : defaultState());
   124→      }
   125→      if (alive) setHydrated(true);
   126→    });
   127→    return () => { alive = false; };
   128→  }, []);
   129→
   130→  // minute ticker: keeps the theme engine and scheduler honest without tick-count timers
   131→  useEffect(() => {
   132→    const t = setInterval(() => setNow(new Date()), 30_000);
   133→    return () => clearInterval(t);
   134→  }, []);
   135→
   136→  // Theme + density classes must live on <html> so that font-size (rem scale),
   137→  // body background and touch-target vars resolve globally (chunk-1 bug fix —
   138→  // keep the sync on documentElement, not just the Shell wrapper).
   139→  useEffect(() => {
   140→    const root = document.documentElement;
   141→    root.classList.remove(
   142→      'theme-morning', 'theme-evening', 'theme-night', 'night',
   143→      'density-young', 'density-school', 'density-adult', 'calm'
   144→    );
   145→    root.classList.add(`theme-${theme}`, `density-${density}`);
   146→    if (theme === 'night') root.classList.add('night');
   147→    if (calm) root.classList.add('calm');
   148→  }, [theme, density, calm]);
   149→
   150→  // Language class on <html> pairs Noto Sans Devanagari with the brand fonts
   151→  // and keeps assistive tech informed (chunk 4B i18n).
   152→  useEffect(() => {
   153→    const root = document.documentElement;
   154→    root.classList.remove('lang-hi', 'lang-hinglish', 'lang-en');
   155→    root.classList.add(`lang-${lang}`);
   156→    root.setAttribute('lang', lang === 'hi' ? 'hi' : 'en');
   157→  }, [lang]);
   158→
   159→  // ---- chunk 5A: account probe — AFTER hydration, never blocking ----
   160→  // The app boots fully from local data; /api/auth/me and /api/capabilities
   161→  // are tried afterwards. Unreachable server => honest 'unavailable' state,
   162→  // every existing feature keeps working (degradation matrix master 8.2).
   163→  useEffect(() => {
   164→    if (!hydrated) return;
   165→    let alive = true;
   166→    void (async () => {
   167→      const caps = await authApi.capabilities();
   168→      const me = await authApi.me();
   169→      if (!alive) return;
   170→      const unavailable = !caps.ok || !me.ok || me.data.unavailable === true;
   171→      setAccountAvail(unavailable ? 'unavailable' : 'ok');
   172→      if (caps.ok) {
   173→        setDevOutbox(caps.data.devOutbox === true);
   174→        const mode = caps.data.ai?.mode;
   175→        setAiMode(mode === 'fake' || mode === 'openai-compatible' ? mode : 'off');
   176→      } else {
   177→        setAiMode('off');
   178→      }
   179→      if (me.ok && me.data.signedIn && me.data.user) {
   180→        setAccountUser(me.data.user);
   181→        await dbMetaSet('account-user-id', me.data.user.id);
   182→      } else {
   183→        setAccountUser(null);
   184→      }
   185→    })();
   186→    return () => { alive = false; };
   187→  }, [hydrated]);
   188→
   189→  const recordAccountId = (id: string) => {
   190→    try {
   191→      localStorage.setItem('lumio-account-last', id);
   192→    } catch {
   193→      /* private mode */
   194→    }
   195→  };
   196→
   197→  const handleSignedIn = (user: AccountUser) => {
   198→    const prevId = (() => {
   199→      try {
   200→        return localStorage.getItem('lumio-account-last');
   201→      } catch {
   202→        return null;
   203→      }
   204→    })();
   205→    const hasLocalProgress = !!stateRef.current?.profile;
   206→    if (prevId && prevId !== user.id && hasLocalProgress) {
   207→      // Never merge silently (spec 8): ask the learner what to do.
   208→      setDevicePrompt({ userId: user.id, userName: user.name });
   209→      setAccountUser(user);
   210→      void dbMetaSet('account-user-id', user.id);
   211→      return;
   212→    }
   213→    recordAccountId(user.id);
   214→    setAccountUser(user);
   215→    void dbMetaSet('account-user-id', user.id);
   216→    toast({ title: tr('account.welcome'), description: tr('account.signedInToast') });
   217→  };
   218→
   219→  const handleSignedOut = () => {
   220→    setAccountUser(null);
   221→    void dbMetaSet('account-user-id', '');
   222→    toast({ title: tr('account.signedOutToast') });
   223→  };
   224→
   225→  const handleDeleted = () => {
   226→    setAccountUser(null);
   227→    try {
   228→      localStorage.removeItem('lumio-account-last');
   229→    } catch {
   230→      /* private mode */
   231→    }
   232→    void dbMetaSet('account-user-id', '');
   233→    toast({ title: tr('account.deletedToast') });
   234→  };
   235→
   236→  // ---- chunk 5B Tier 2: AI opt-in toggle support ----
   237→  // Settings calls aiApi.enable itself, then asks LumioApp to mirror the flag
   238→  // locally (instant UI) and re-probe /api/auth/me (source of truth).
   239→  const refreshAccount = useCallback(async () => {
   240→    const me = await authApi.me();
   241→    if (me.ok && me.data.signedIn && me.data.user) setAccountUser(me.data.user);
   242→    else setAccountUser(null);
   243→  }, []);
   244→
   245→  const aiToggled = (next: boolean) => {
   246→    setAccountUser((u) => (u ? { ...u, aiEnabled: next } : u));
   247→    void refreshAccount();
   248→  };
   249→
   250→  const accountSlot = (
   251→    <AccountBadge
   252→      lang={lang}
   253→      user={accountUser}
   254→      avail={accountAvail}
   255→      devOutbox={devOutbox}
   256→      onSignedIn={handleSignedIn}
   257→      onSignedOut={handleSignedOut}
   258→      onDeleted={handleDeleted}
   259→    />
   260→  );
   261→
   262→  // chunk 5B: when a chapter is open in Reader the panel grounds on THAT chapter;
   263→  // otherwise StudyBuddy shows a chapter select (chapter = null).
   264→  const studyBuddyChapter: BuddyChapter | null =
   265→    reader !== null && state?.profile?.band
   266→      ? (() => {
   267→          const chapters = BANDS[state.profile!.band].chapters;
   268→          const ch = chapters[Math.min(Math.max(0, reader), chapters.length - 1)];
   269→          return ch
   270→            ? { id: `${state.profile!.band}#${reader}`, title: ch.title, subject: ch.subject, chapterIndex: reader }
   271→            : null;
   272→        })()
   273→      : null;
   274→  const signedIn = !!accountUser;
   275→  const aiEnabled = accountUser?.aiEnabled === true;
   276→  const aiDemo = aiMode === 'fake';
   277→
   278→  // ---- chunk 6: syllabus tracker (CBSE Class 11 gate; derived, never stored) ----
   279→  const syllabusAll = state?.profile ? syllabusForProfile(state.profile.className, state.profile.board) : null;
   280→  const syllabusAvailable = !!syllabusAll;
   281→  const syllabusSubjects = (syllabusAll ?? []).filter(
   282→    (s) => !state?.syllabus.hiddenSubjects.includes(s.id)
   283→  );
   284→  const todayKey = dayKey(now, tz);
   285→  const acceptedSyllabusPlan: AcceptedPlan | null = state?.syllabus.plans[todayKey] ?? null;
   286→  const dueCardsCount = Object.values(state?.srs ?? {}).filter((c) => c.due <= todayKey).length;
   287→
   288→  /** Slots for today (from the existing planner, overrides respected). */
   289→  const syllabusSlots = state?.plan ? planDayWithOverrides(state.plan, state.planOverrides[todayKey]).slots : [];
   290→
   291→  /** Nearest future exam date the learner gave us — one date for all subjects. */
   292→  const syllabusExamDate = (() => {
   293→    const p = state?.profile;
   294→    const candidates = [p?.nextExamDate, p?.examDate, p?.compExamDate, p?.targetDate].filter(
   295→      (d): d is string => !!d
   296→    );
   297→    return candidates.sort()[0] ?? undefined;
   298→  })();
   299→
   300→  const examDatesForSyllabus: Record<string, string | undefined> = {};
   301→  for (const s of syllabusSubjects) examDatesForSyllabus[s.id] = syllabusExamDate;
   302→
   303→  /** Topics already planned or marked today are skipped by the planner. */
   304→  const plannedTodaySet: ReadonlySet<TopicId> = new Set([
   305→    ...(acceptedSyllabusPlan?.items.flatMap((i) => i.topicIds) ?? []),
   306→    ...Object.entries(state?.syllabus.topics ?? [])
   307→      .filter(([, rec]) => rec.updatedOn === todayKey)
   308→      .map(([id]) => id),
   309→  ]);
   310→
   311→  const buildSyllabusPlanToday = (): SyllabusPlanT =>
   312→    buildSyllabusPlan({
   313→      today: todayKey,
   314→      dayIndex: 0,
   315→      slots: syllabusSlots,
   316→      subjects: syllabusSubjects,
   317→      compSub: state?.profile?.compSub,
   318→      progress: state?.syllabus.topics ?? {},
   319→      examDates: examDatesForSyllabus,
   320→      plannedToday: plannedTodaySet,
   321→      dueCardsCount,
   322→    });
   323→
   324→  const deterministicSyllabusPlan: SyllabusPlanT = buildSyllabusPlanToday();
   325→
   326→  // service-worker update prompt (chunk 2): never auto-swaps mid-session
   327→  useEffect(() => {
   328→    if (typeof navigator === 'undefined' || !('serviceWorker' in navigator)) return;
   329→    let reg: ServiceWorkerRegistration | null = null;
   330→    const findWaiting = () => {
   331→      if (reg?.waiting && navigator.serviceWorker.controller) setSwWaiting(reg.waiting);
   332→    };
   333→    navigator.serviceWorker.register('/sw.js')
   334→      .then((r) => {
   335→        reg = r;
   336→        findWaiting();
   337→        r.addEventListener('updatefound', () => {
   338→          const nw = r.installing;
   339→          nw?.addEventListener('statechange', () => {
   340→            if (nw.state === 'installed') findWaiting();
   341→          });
   342→        });
   343→      })
   344→      .catch(() => { /* offline/PWA unsupported — the app still works */ });
   345→  }, []);
   346→
   347→  /**
   348→   * The one mutation path: append an event, reapply the projection, mirror the
   349→   * snapshot. Derived values (XP, best scores, streak, achievements) come out
   350→   * of the projection only — never set by callers (5.4).
   351→   */
   352→  const dispatch = useCallback((e: LumioEvent) => {
   353→    const prev = stateRef.current;
   354→    if (!prev) return;
   355→    const zone = prev.profile?.tz ?? DEFAULT_TZ;
   356→    const next = projectState(applyEvent(prev, e), new Date(), zone);
   357→    stateRef.current = next;
   358→    setState(next);
   359→    const res = saveState(next);
   360→    setSaveError(res.ok ? undefined : res.error);
   361→    // Streak Saver notifications (chunk 4C): earned / bought / used.
   362→    const prevOwned = prev.freeze.owned;
   363→    const nextOwned = next.freeze.owned;
   364→    if (nextOwned > prevOwned) {
   365→      toast({
   366→        title: e.type === 'saver-bought' ? 'Streak Saver purchased' : 'Streak Saver earned!',
   367→        description:
   368→          e.type === 'saver-bought'
   369→            ? 'It will protect your streak if a day is missed.'
   370→            : 'Great performance — your streak just got an extra life.',
   371→      });
   372→    } else if (nextOwned < prevOwned) {
   373→      toast({
   374→        title: 'Streak Saver used',
   375→        description: 'Your streak stays alive. Keep learning to earn more.',
   376→      });
   377→    }
   378→    void appendEvent(e, zone).catch(() => {
   379→      // event log unavailable (private mode) — the snapshot mirror keeps the
   380→      // session working; the learner sees an honest error if saving failed.
   381→    });
   382→  }, []);
   383→
   384→  // ---- mutations ----
   385→  const finishFirstRun = (r: FirstRunResult) => {
   386→    dispatch({ type: 'onboarding-finished', profile: r.profile, plan: r.plan });
   387→    setEditProfile(false);
   388→    setScreen('home');
   389→  };
   390→
   391→  const saveDraft = (draft: WizardDraft | null) => {
   392→    dispatch({ type: 'draft-saved', draft: draft as unknown as Record<string, unknown> | null });
   393→  };
   394→
   395→  const patchSettings = (patch: Partial<Settings>) => {
   396→    dispatch({ type: 'settings-changed', patch });
   397→  };
   398→
   399→  const patchPlan = (patch: Partial<NonNullable<LumioState['plan']>>) => {
   400→    if (!state?.plan) return;
   401→    dispatch({ type: 'plan-changed', plan: { ...state.plan, ...patch } });
   402→  };
   403→
   404→  const markPlanReviewed = () => {
   405→    if (!state) return;
   406→    const today = dayKey(new Date(), tz);
   407→    if (state.planReviews[today]) return;
   408→    // The activity makes the daily quest claimable; the Claim button grants the XP.
   409→    dispatch({ type: 'plan-reviewed', date: today });
   410→  };
   411→
   412→  const claimQuest = (questId: string, periodKey: string, xp: number) => {
   413→    if (!state) return;
   414→    if (state.quests.some((c) => c.questId === questId && c.periodKey === periodKey)) return;
   415→    dispatch({ type: 'quest-claimed', questId, periodKey, xp, date: dayKey(new Date(), tz) });
   416→  };
   417→
   418→  const finishQuiz = (chapterIndex: number, outcome: QuizOutcome) => {
   419→    dispatch({
   420→      type: 'quiz-attempted',
   421→      chapterId: `${state?.profile?.band ?? 'sec'}#${chapterIndex}`,
   422→      score: outcome.score,
   423→      nq: outcome.nq,
   424→      wrongIds: outcome.wrongIds,
   425→      theme,
   426→      date: dayKey(new Date(), tz),
   427→    });
   428→    setQuiz(null);
   429→    setReader(null);
   430→    setScreen('home');
   431→  };
   432→
   433→  const flashcardsReviewed = (chapterIndex: number) => {
   434→    dispatch({ type: 'flashcards-reviewed', chapterId: `${state?.profile?.band ?? 'sec'}#${chapterIndex}`, date: dayKey(new Date(), tz) });
   435→  };
   436→
   437→  // ---- chunk 3: games, labs and the practice engine ----
   438→  const winGame = (gameId: string) => {
   439→    const prev = stateRef.current;
   440→    const firstClear = !prev?.xpLedger.some((x) => x.source === 'game' && x.ref === gameId);
   441→    dispatch({
   442→      type: 'game-won',
   443→      gameId,
   444→      xp: GAME_XP[gameId] ?? 20,
   445→      date: dayKey(new Date(), tz),
   446→      lab: LAB_GAME_IDS.has(gameId),
   447→      firstClear,
   448→    });
   449→  };
   450→  const reviewCard = (cardId: string, correct: boolean) => {
   451→    dispatch({ type: 'card-reviewed', cardId, correct, date: dayKey(new Date(), tz) });
   452→  };
   453→  const finishPractice = (kind: PracticeKind) => {
   454→    dispatch({ type: 'practice-done', kind, xp: PRACTICE_XP[kind] ?? 10, date: dayKey(new Date(), tz) });
   455→  };
   456→  const finishLegendary = () => {
   457→    dispatch({ type: 'legendary-completed', date: dayKey(new Date(), tz) });
   458→  };
   459→  const tagMistake = (questionId: string, tag: MistakeTag | null) => {
   460→    dispatch({ type: 'mistake-tagged', questionId, tag });
   461→  };
   462→
   463→  // ---- chunk 5B: AI practice-only saves (kind 'notes' | 'practice') ----
   464→  // Never touches XP/streak/mastery — the projection keeps the record in
   465→  // state.aiContent (deduped by id, capped at 30, capped by applyEvent).
   466→  const saveAiContent = (
   467→    kind: 'notes' | 'practice',
   468→    id: string,
   469→    date: string,
   470→    payload: NotesAiResult | { items: PracticeAiItem[] },
   471→    label: string,
   472→  ) => {
   473→    dispatch({ type: 'ai-generated', id, kind, date, payload, label });
   474→  };
   475→  const saveAiNotes = (args: {
   476→    id: string;
   477→    kind: 'notes';
   478→    date: string;
   479→    payload: NotesAiResult;
   480→    label: string;
   481→  }) => {
   482→    saveAiContent(args.kind, args.id, args.date, args.payload, args.label);
   483→  };
   484→
   485→  // ---- chunk 4A: focus suite, mocks, streak freezes, reschedule ----
   486→  const finishFocusBlock = (mins: number) => {

   634→        slotIndex: it.slotIndex,
   635→        subjectId: slotSubjectOf.get(it.slotIndex) ?? 'maths',
   636→        topicIds: it.topicIds,
   637→        topicTitles: it.topicIds.map((id) => titleOf.get(id) ?? id),
