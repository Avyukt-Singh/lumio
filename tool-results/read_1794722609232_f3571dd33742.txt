   579→/** Strict fallback chains (chunk 4B): hi -> hi-Latn -> en; hinglish -> en. */
   580→export function t(lang: UiLang, key: DictKey): string {
   581→  if (lang === 'hi') return HI[key] ?? HINGLISH[key] ?? EN[key];
   582→  if (lang === 'hinglish') return HINGLISH[key] ?? EN[key];
   583→  return EN[key];
   584→}
   585→
   586→/** Convenience: translate for a LumioState profile language (defaults to English). */
   587→export function tFor(lang: UiLang | undefined | null): (key: DictKey) => string {
   588→  return (key: DictKey) => t(lang ?? 'en', key);
   589→}
   590→
   591→export function isUiLang(v: unknown): v is UiLang {
   592→  return v === 'en' || v === 'hi' || v === 'hinglish';
   593→}

   595→/** 'daily-plan-review' -> DictKey 'quest.dailyPlanReview.title' | '.desc' */
   596→function questDictKey(questId: string, kind: 'title' | 'desc'): DictKey {
   597→  const camel = questId
   598→    .split('-')
   599→    .map((part, i) => (i === 0 ? part : part.charAt(0).toUpperCase() + part.slice(1)))
   600→    .join('');
   601→  return `quest.${camel}.${kind}` as DictKey;
   602→}
   603→
   604→export function questTitle(lang: UiLang, questId: string, fallback: string): string {
   605→  const key = questDictKey(questId, 'title');
   606→  return lang === 'en' ? fallback : t(lang, key) || fallback;
   607→}
   608→
   609→export function questDesc(lang: UiLang, questId: string, fallback: string): string {
   610→  const key = questDictKey(questId, 'desc');
   611→  return lang === 'en' ? fallback : t(lang, key) || fallback;
   612→}
