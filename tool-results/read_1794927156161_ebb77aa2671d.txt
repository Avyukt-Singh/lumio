   109→/** Deterministic Fisher-Yates so repeats across rounds vary with the seed. */
   110→function seededShuffle<T>(items: T[], seed: number): T[] {
   111→  const arr = [...items];
   112→  let s = seed >>> 0 || 1;
   113→  const rand = () => {
   114→    s = (s * 1664525 + 1013904223) >>> 0;
   115→    return s / 4294967296;
   116→  };
   117→  for (let i = arr.length - 1; i > 0; i--) {
   118→    const j = Math.floor(rand() * (i + 1));
   119→    [arr[i], arr[j]] = [arr[j], arr[i]];
   120→  }
   121→  return arr;
   122→}

   124→/** Resolve stable question ids (`${band}#${ch}#${q}`) against the band content. */
   125→function resolveMissed(band: Band, wrong: Record<string, 1>): MissedItem[] {
   126→  const out: MissedItem[] = [];
   127→  for (const id of Object.keys(wrong)) {
   128→    const parts = id.split('#');
   129→    if (parts.length !== 3 || parts[0] !== band) continue;
   130→    const ci = Number(parts[1]);
   131→    const qi = Number(parts[2]);
   132→    if (!Number.isFinite(ci) || !Number.isFinite(qi)) continue;
   133→    const ch = BANDS[band].chapters[ci];
   134→    const q = ch?.questions[qi];
   135→    if (!ch || !q) continue;
   136→    out.push({ id, chapterTitle: ch.title, subject: ch.subject, q });
   137→  }
   138→  return out;
   139→}

   254→function FlashcardsTab({
   255→  state,
   256→  today,
   257→  band,
   258→  onCardReview,
   259→  onPracticeDone,
   260→}: {
   261→  state: LumioState;
   262→  today: string;
   263→  band: Band;
   264→  onCardReview: (cardId: string, correct: boolean) => void;
   265→  onPracticeDone: (kind: PracticeKind) => void;
   266→}) {
   267→  const deck = BANDS[band].chapters.flatMap((ch, i) => flashcardsFor(band, i, ch));
   268→  const due = deck.filter((c) => {
   269→    const s = state.srs[c.id];
   270→    return !s || s.due <= today;
   271→  });
   272→  const boxCounts = [1, 2, 3, 4, 5].map((b) => deck.filter((c) => state.srs[c.id]?.box === b).length);
   273→  const unseen = deck.filter((c) => !state.srs[c.id]).length;
   274→  const nextDue = deck.reduce<string | null>((min, c) => {
   275→    const d = state.srs[c.id]?.due;
   276→    if (d && d > today && (min === null || d < min)) return d;
   277→    return min;
   278→  }, null);
   279→  const collected = practiceDoneToday(state, 'srs', today);
   280→
   281→  const [queue, setQueue] = useState<Flashcard[] | null>(null);
   282→  const [pos, setPos] = useState(0);
   283→  const [flipped, setFlipped] = useState(false);
   284→  const [reviewed, setReviewed] = useState(0);
   285→  const finishedRef = useRef(false);
   286→
   287→  const startSession = () => {
   288→    if (due.length === 0) return;
   289→    finishedRef.current = false;
   290→    setQueue(due);
   291→    setPos(0);
   292→    setFlipped(false);
   293→    setReviewed(0);
   294→  };
   295→
   296→  const endSession = () => {
   297→    setQueue(null);
   298→    setFlipped(false);
   299→  };
   300→
   301→  const review = (correct: boolean) => {
   302→    if (!queue || pos >= queue.length) return;
   303→    const card = queue[pos];
   304→    onCardReview(card.id, correct);
   305→    setFlipped(false);
   306→    setReviewed((r) => r + 1);
   307→    const np = pos + 1;
   308→    setPos(np);
   309→    if (np >= queue.length && !finishedRef.current) {
   310→      finishedRef.current = true;
   311→      onPracticeDone('srs');
   312→    }
   313→  };
   314→
   315→  // Overview
   316→  if (queue === null) {
   317→    return (
   318→      <TabCard>
   319→        <h2 className="font-heading text-lg font-semibold">Flashcards</h2>
   320→        <p className="mt-1 text-sm text-muted-foreground">
   321→          {due.length === 1 ? '1 card due today' : `${due.length} cards due today`}
   322→        </p>
   323→        <p className="mt-1 text-xs text-muted-foreground">
   324→          Box 1: {boxCounts[0]} &middot; Box 2: {boxCounts[1]} &middot; Box 3: {boxCounts[2]} &middot;
   325→          Box 4: {boxCounts[3]} &middot; Box 5: {boxCounts[4]} &middot; Not started: {unseen}
   326→        </p>
   327→
   328→        {due.length === 0 ? (
   329→          <div className="mt-2">
   330→            <EmptyState note="Nothing due right now — cards return after 1, 2, 4, 8 or 16 days depending on the box they reach." />
   331→            {nextDue !== null && (
   332→              <p className="text-center text-sm text-muted-foreground">
   333→                Next cards return on {nextDue}.
   334→              </p>
   335→            )}
   336→          </div>
   337→        ) : (
   338→          <div className="mt-4">
   339→            <Button className="min-h-[44px]" onClick={startSession}>
   340→              Start review session
   341→            </Button>
   342→          </div>
   343→        )}
   344→
   345→        <p className="mt-4 text-xs text-muted-foreground">
   346→          New cards appear as you study chapters.
   347→        </p>
   348→      </TabCard>
   349→    );
   350→  }
   351→
   352→  // Session summary
   353→  if (pos >= queue.length) {
   354→    return (
   355→      <TabCard>
   356→        <div className="flex flex-col items-center gap-3 py-4 text-center">
   357→          <Mascot pose="cheering" size="xl" bob />
   358→          <h2 className="font-heading text-xl font-bold">
   359→            Reviewed {reviewed} card{reviewed === 1 ? '' : 's'}
   360→          </h2>
   361→          <p className="text-sm text-muted-foreground">
   362→            +{PRACTICE_XP.srs} XP{' '}
   363→            {collected ? 'collected today' : "for today's first review set"}
   364→          </p>
   365→          <Button variant="outline" className="min-h-[44px]" onClick={endSession}>
   366→            Back to overview
   367→          </Button>
   368→        </div>
   369→      </TabCard>
   370→    );
   371→  }
   372→
   373→  // One card at a time
   374→  const card = queue[pos];
   375→  return (
   376→    <TabCard>
   377→      <div className="flex items-center justify-between gap-3">
   378→        <h2 className="font-heading text-lg font-semibold">
   379→          Card {pos + 1} of {queue.length}
   380→        </h2>
   381→        <Button variant="ghost" className="min-h-[44px]" onClick={endSession}>
   382→          End session
   383→        </Button>
   384→      </div>
   385→      <Progress
   386→        value={(pos / queue.length) * 100}
   387→        className="mt-2 h-2"
   388→        aria-label={`Review progress: card ${pos + 1} of ${queue.length}`}
   389→      />
   390→
   391→      <div className="mt-4 rounded-2xl border border-border bg-background p-5 text-center">
   392→        <Badge variant="secondary">{card.kind === 'qa' ? 'Question' : 'Term'}</Badge>
   393→        <p className="mt-3 font-heading text-lg font-semibold sm:text-xl">{card.front}</p>
   394→        {flipped && (
   395→          <div
   396→            role="status"
   397→            aria-live="polite"
   398→            className="anim-pop mt-4 rounded-xl bg-success-soft p-4 text-success-soft-foreground"
   399→          >
   400→            <p className="text-sm font-bold">Answer</p>
   401→            <p className="mt-1 text-sm">{card.back}</p>
   402→          </div>
   403→        )}
   404→      </div>
   405→
   406→      <div className="mt-4 flex flex-wrap items-center justify-center gap-2">
   407→        {!flipped ? (
   408→          <Button className="min-h-[44px]" onClick={() => setFlipped(true)}>
   409→            <Eye className="size-4" aria-hidden="true" /> Flip card
   410→          </Button>
   411→        ) : (
   412→          <>
   413→            <Button
   414→              className="min-h-[44px] bg-success text-success-foreground hover:bg-success/90"
   415→              onClick={() => review(true)}
   416→            >
   417→              <Check className="size-4" aria-hidden="true" /> I knew it
   418→            </Button>
   419→            <Button variant="outline" className="min-h-[44px]" onClick={() => review(false)}>
   420→              See it again
   421→            </Button>
   422→          </>
   423→        )}
   424→      </div>
   425→    </TabCard>
   426→  );
   427→}

   481→function DrillTab({
   482→  state,
   483→  today,
   484→  band,
   485→  onPracticeDone,
   486→}: {
   487→  state: LumioState;
   488→  today: string;
   489→  band: Band;
   490→  onPracticeDone: (kind: PracticeKind) => void;
   491→}) {
   492→  const pool = BANDS[band].chapters.flatMap((ch, ci) =>
   493→    ch.questions.map((q, qi) => ({ key: `${band}#${ci}#${qi}`, q, chapterTitle: ch.title })),
   494→  );
   495→
   496→  const [phase, setPhase] = useState<'idle' | 'run' | 'done'>('idle');
   497→  const [untimed, setUntimed] = useState(false);
   498→  const [order, setOrder] = useState<number[] | null>(null);
   499→  const [pos, setPos] = useState(0);
   500→  const [score, setScore] = useState(0);
   501→  const [answered, setAnswered] = useState(0);
   502→  const [picked, setPicked] = useState<number | null>(null);
   503→  const [secondsLeft, setSecondsLeft] = useState<number | null>(null);
   504→  const [deadline, setDeadline] = useState<number | null>(null);
   505→  const [announce, setAnnounce] = useState('');
   506→
   507→  const endedRef = useRef(false);
   508→  const announcedRef = useRef<Set<number>>(new Set());
   509→  const timeoutRef = useRef<number | null>(null);
   510→  const endRoundRef = useRef<() => void>(() => {});
   511→
   512→  // Derived values (declared above all effects, per project rule).
   513→  const running = phase === 'run' && order !== null;
   514→  const cur = order !== null ? pool[order[pos] % pool.length] : null;
   515→  const collected = practiceDoneToday(state, 'drill', today);
   516→
   517→  const endRound = () => {
   518→    if (endedRef.current) return;
   519→    endedRef.current = true;
   520→    if (timeoutRef.current !== null) {
   521→      window.clearTimeout(timeoutRef.current);
   522→      timeoutRef.current = null;
   523→    }
   524→    setPicked(null);
   525→    setPhase('done');
   526→    onPracticeDone('drill');
   527→  };
   528→  endRoundRef.current = endRound;
   529→
   530→  const finishAnswer = () => {
   531→    if (timeoutRef.current !== null) {
   532→      window.clearTimeout(timeoutRef.current);
   533→      timeoutRef.current = null;
   534→    }
   535→    if (endedRef.current) return;
   536→    const na = answered + 1;
   537→    if (untimed && na >= DRILL_UNTIMED_COUNT) {
   538→      setAnswered(na);
   539→      endRound();
   540→      return;
   541→    }
   542→    setAnswered(na);
   543→    setPicked(null);
   544→    setPos((p) => (p + 1) % Math.max(1, order?.length ?? 1));
   545→  };
   546→
   547→  const pick = (i: number) => {
   548→    if (!running || picked !== null || !cur) return;
   549→    const correct = i === cur.q.answer;
   550→    setPicked(i);
   551→    if (correct) {
   552→      setScore((s) => s + 1);
   553→      finishAnswer();
   554→    } else {
   555→      // Wrong: show the explanation briefly, then move on.
   556→      timeoutRef.current = window.setTimeout(finishAnswer, 1600);
   557→    }
   558→  };
   559→
   560→  const startRound = () => {
   561→    endedRef.current = false;
   562→    announcedRef.current = new Set();
   563→    const seed = (Date.now() % 2147483647) || 1;
   564→    setOrder(seededShuffle(pool.map((_, i) => i), seed));
   565→    setPos(0);
   566→    setScore(0);
   567→    setAnswered(0);
   568→    setPicked(null);
   569→    setSecondsLeft(untimed ? null : Math.ceil(DRILL_MS / 1000));
   570→    setDeadline(untimed ? null : Date.now() + DRILL_MS);
   571→    setAnnounce(untimed ? 'Untimed round started. Ten questions.' : 'Round started. Sixty seconds.');
   572→    setPhase('run');
   573→  };
   574→
   575→  // Countdown: ticks every 250ms but re-renders only when the second changes.
   576→  useEffect(() => {
   577→    if (!running || deadline === null) return;
   578→    const t = window.setInterval(() => {
   579→      const s = Math.ceil((deadline - Date.now()) / 1000);
   580→      setSecondsLeft((prev) => (prev === s ? prev : s));
   581→      if (s <= 0) {
   582→        window.clearInterval(t);
   583→        endRoundRef.current();
   584→      }
   585→    }, 250);
   586→    return () => window.clearInterval(t);
   587→  }, [running, deadline]);
   588→
   589→  // Announce time politely only at 30s and 10s — never every second.
   590→  useEffect(() => {
   591→    if (!running || secondsLeft === null) return;
   592→    if ((secondsLeft === 30 || secondsLeft === 10) && !announcedRef.current.has(secondsLeft)) {
   593→      announcedRef.current.add(secondsLeft);
   594→      setAnnounce(`${secondsLeft} seconds left.`);
   595→    }
   596→  }, [running, secondsLeft]);
   597→
   598→  useEffect(
   599→    () => () => {
   600→      if (timeoutRef.current !== null) window.clearTimeout(timeoutRef.current);
   601→    },
   602→    [],
   603→  );
   604→
   605→  if (phase === 'idle') {
   606→    return (
   607→      <TabCard>
   608→        <h2 className="font-heading text-lg font-semibold">Timed drill</h2>
   609→        <p className="mt-1 text-sm text-muted-foreground">
   610→          A 60-second rapid-fire round across every chapter in this band. A wrong answer shows the
   611→          explanation for a moment, then moves on. The pool loops if you outrun it.
   612→        </p>
   613→        <div className="mt-4 flex flex-wrap items-center gap-3">
   614→          <Button className="min-h-[44px]" onClick={startRound}>
   615→            <Timer className="size-4" aria-hidden="true" /> Start drill
   616→          </Button>
   617→          <button
   618→            type="button"
   619→            aria-pressed={untimed}
   620→            onClick={() => setUntimed(!untimed)}
   621→            className={cn(
   622→              'press flex min-h-[44px] items-center gap-2 rounded-full border px-4 text-sm font-medium',
   623→              untimed ? 'border-primary bg-primary/10 text-primary' : 'border-border',
   624→            )}
   625→          >
   626→            <Clock className="size-4" aria-hidden="true" />
   627→            Untimed round {untimed ? 'on' : 'off'}
   628→          </button>
   629→        </div>
   630→        <p className="mt-3 text-xs text-muted-foreground">
   631→          Untimed rounds run 10 questions with no clock.
   632→        </p>
   633→      </TabCard>
   634→    );
   635→  }
   636→
   637→  if (phase === 'done') {
   638→    return (
   639→      <TabCard>
   640→        <div className="flex flex-col items-center gap-3 py-4 text-center">
   641→          <Mascot pose="excited" size="xl" bob />
   642→          <h2 className="font-heading text-xl font-bold">{score} correct</h2>
   643→          <p className="text-sm text-muted-foreground">
   644→            {answered} question{answered === 1 ? '' : 's'} answered this round.
   645→          </p>
   646→          <p className="text-sm text-muted-foreground">
   647→            +{PRACTICE_XP.drill} XP {collected ? 'collected today' : "for today's first round"}
   648→          </p>
   649→          <Button className="min-h-[44px]" onClick={() => setPhase('idle')}>
   650→            New round
   651→          </Button>
   652→        </div>
   653→      </TabCard>
   654→    );
   655→  }
   656→
   657→  // Running
   658→  if (!cur) return null;
   659→  return (
   660→    <TabCard>
   661→      <div className="flex flex-wrap items-center justify-between gap-2">
   662→        {secondsLeft !== null ? (
   663→          <Badge variant="secondary" className="font-mono text-base">
   664→            {secondsLeft}s left
   665→          </Badge>
   666→        ) : (
   667→          <Badge variant="secondary">Untimed</Badge>
   668→        )}
   669→        <p className="text-sm text-muted-foreground">
   670→          Score {score} &middot;{' '}
   671→          {untimed
   672→            ? `${Math.min(answered + 1, DRILL_UNTIMED_COUNT)} of ${DRILL_UNTIMED_COUNT}`
   673→            : `${answered} answered`}
   674→        </p>
   675→      </div>
   676→      {untimed && (
   677→        <Progress
   678→          value={(answered / DRILL_UNTIMED_COUNT) * 100}
   679→          className="mt-2 h-2"
   680→          aria-label={`Drill progress: ${answered} of ${DRILL_UNTIMED_COUNT} questions answered`}
   681→        />
   682→      )}
   683→      <p aria-live="polite" className="mt-1 min-h-[1rem] text-xs text-muted-foreground">
   684→        {announce}
   685→      </p>
   686→
   687→      <p className="mt-2 font-heading text-base font-semibold sm:text-lg">{cur.q.text}</p>
   688→      <p className="mt-1 text-xs text-muted-foreground">{cur.chapterTitle}</p>

   695→            index={i}
   696→            state={

   698→                ? 'idle'
   699→                : picked === i

   703→                  : 'dim'
   704→            }
   705→            onClick={() => pick(i)}

   844→        value={(pos / ADAPTIVE_COUNT) * 100}
   845→        className="mt-2 h-2"

  1028→function RevisionTab({
  1029→  state,
  1030→  today,
  1031→  band,
  1032→  onPracticeDone,
  1033→}: {
  1034→  state: LumioState;
  1035→  today: string;
  1036→  band: Band;
  1037→  onPracticeDone: (kind: PracticeKind) => void;
  1038→}) {
  1039→  const chapters = BANDS[band].chapters;
  1040→  const completed = chapters
  1041→    .map((_, i) => i)
  1042→    .filter((i) => (state.res[chapterId(band, i)]?.best ?? 0) >= 1);
  1043→  const locked = completed.length < 2;
  1044→  const collected = practiceDoneToday(state, 'revision', today);
  1045→
  1046→  const [round, setRound] = useState(0);
  1047→
  1048→  const items: QuizItem[] = [];
  1049→  if (!locked) {
  1050→    const perCh = Math.ceil(REVISION_COUNT / completed.length);
  1051→    const queues = completed.map((ci, k) => {
  1052→      const qs = chapters[ci].questions;
  1053→      const expanded: Array<{ q: Question; qi: number }> = [];
  1054→      for (let n = 0; expanded.length < perCh; n++) {
  1055→        const qi = n % qs.length;
  1056→        expanded.push({ q: qs[qi], qi });
  1057→      }
  1058→      return seededShuffle(expanded, (round + 1) * 97 + k * 13 + 7).map(({ q, qi }) => ({
  1059→        key: `${band}#${ci}#${qi}`,
  1060→        q,
  1061→        chapterTitle: chapters[ci].title,
  1062→      }));
  1063→    });
  1064→    for (let r = 0; items.length < REVISION_COUNT; r++) {
  1065→      let any = false;
  1066→      for (const queue of queues) {
  1067→        if (items.length >= REVISION_COUNT) break;
  1068→        const it = queue[r];
  1069→        if (it) {
  1070→          items.push(it);
  1071→          any = true;
  1072→        }
  1073→      }
  1074→      if (!any) break;
  1075→    }
  1076→  }
  1077→
  1078→  if (locked) {
  1079→    return (
  1080→      <TabCard>
  1081→        <h2 className="font-heading text-lg font-semibold">Mixed revision</h2>
  1082→        <EmptyState note="Complete at least two chapter quizzes first — revision mixes what you have already learned." />
  1083→        <p className="text-center text-sm text-muted-foreground">
  1084→          Chapters completed so far: {completed.length}
  1085→        </p>
  1086→      </TabCard>
  1087→    );
  1088→  }
  1089→
  1090→  return (
  1091→    <div className="flex flex-col gap-3">
  1092→      <TabCard>
  1093→        <h2 className="font-heading text-lg font-semibold">Mixed revision</h2>
  1094→        <p className="mt-1 text-sm text-muted-foreground">
  1095→          Ten questions interleaved across your {completed.length} completed chapter
  1096→          {completed.length === 1 ? '' : 's'} — revising in mixed order is harder and it sticks
  1097→          better.
  1098→        </p>
  1099→      </TabCard>
  1100→      <RunQuiz
  1101→        key={round}
  1102→        items={items}
  1103→        doneTitle="Mixed revision complete."
  1104→        doneXpLine={`+${PRACTICE_XP.revision} XP ${
  1105→          collected ? 'collected today' : "for today's first revision"
  1106→        }`}
  1107→        onDone={() => onPracticeDone('revision')}
  1108→        onAgain={() => setRound((r) => r + 1)}
  1109→      />
  1110→    </div>
  1111→  );
  1112→}

  1116→function LegendaryTab({
  1117→  state,
  1118→  band,
  1119→  onLegendary,
  1120→}: {
  1121→  state: LumioState;
  1122→  band: Band;
  1123→  onLegendary: () => void;
  1124→}) {
  1125→  const chapters = BANDS[band].chapters;
  1126→  const mastered = masteredCount(state);
  1127→  const locked = mastered < 2;
  1128→  const doneEver = legendaryDone(state);
  1129→
  1130→  const [round, setRound] = useState(0);
  1131→
  1132→  // Hard mix: the last question of each chapter, padded with earlier ones if needed.
  1133→  const items: QuizItem[] = [];
  1134→  if (!locked) {
  1135→    for (let back = 0; items.length < LEGENDARY_COUNT && back < 64; back++) {
  1136→      for (let ci = 0; ci < chapters.length && items.length < LEGENDARY_COUNT; ci++) {
  1137→        const qs = chapters[ci].questions;
  1138→        const qi = qs.length - 1 - back;
  1139→        if (qi >= 0) {
  1140→          items.push({ key: `${band}#${ci}#${qi}`, q: qs[qi], chapterTitle: chapters[ci].title });

  1191→function ErrorLabTab({
  1192→  state,
  1193→  today,
  1194→  band,
  1195→  onPracticeDone,
  1196→}: {
  1197→  state: LumioState;
  1198→  today: string;
  1199→  band: Band;
  1200→  onPracticeDone: (kind: PracticeKind) => void;
  1201→}) {
  1202→  const missed = resolveMissed(band, state.wrong);
  1203→  const collected = practiceDoneToday(state, 'errorlab', today);
  1204→
  1205→  const [pos, setPos] = useState(0);
  1206→  const [done, setDone] = useState(false);
  1207→  const paidRef = useRef(false);
  1208→
  1209→  const advance = () => {
  1210→    if (pos === 0 && !paidRef.current) {
  1211→      paidRef.current = true;
  1212→      onPracticeDone('errorlab');
  1213→    }
  1214→    if (pos + 1 >= missed.length) {
  1215→      setDone(true);
  1216→    } else {
  1217→      setPos(pos + 1);
  1218→    }
  1219→  };
  1220→
  1221→  if (missed.length === 0) {
  1222→    return (
  1223→      <TabCard>
  1224→        <h2 className="font-heading text-lg font-semibold">Error lab</h2>
  1225→        <EmptyState note="No misses recorded yet — the Error Lab fills up as questions come back wrong in quizzes and practice." />
  1226→      </TabCard>
  1227→    );
  1228→  }
  1229→
  1230→  if (done) {
  1231→    return (
  1232→      <TabCard>
  1233→        <div className="flex flex-col items-center gap-3 py-4 text-center">
  1234→          <Mascot pose="excited" size="xl" bob />
  1235→          <h2 className="font-heading text-xl font-bold">
  1236→            Reviewed {missed.length} mistake{missed.length === 1 ? '' : 's'}
  1237→          </h2>
  1238→          <p className="text-sm text-muted-foreground">
  1239→            +{PRACTICE_XP.errorlab} XP {collected ? 'collected today' : "for today's first lab"}
  1240→          </p>
  1241→          <Button
  1242→            variant="outline"
  1243→            className="min-h-[44px]"
  1244→            onClick={() => {
  1245→              setPos(0);
  1246→              setDone(false);
  1247→              paidRef.current = true;
  1248→            }}
  1249→          >
  1250→            Review them again
  1251→          </Button>
  1252→        </div>
  1253→      </TabCard>
  1254→    );
  1255→  }
  1256→
  1257→  const row = missed[pos];
  1258→  const tagged = state.mistakes[row.id]?.tag;
  1259→  const tagLabel = MISTAKE_TAGS.find((t) => t.tag === tagged)?.label;
  1260→  const steps = workedSteps(row.q, row.chapterTitle, row.subject);
  1261→  const last = pos + 1 >= missed.length;
  1262→
  1263→  return (
  1264→    <TabCard>
  1265→      <div className="flex items-center justify-between gap-3">
  1266→        <h2 className="font-heading text-lg font-semibold">
  1267→          Mistake {pos + 1} of {missed.length}
  1268→        </h2>
  1269→        <Badge variant="secondary">{row.subject}</Badge>
  1270→      </div>
  1271→
  1272→      <p className="mt-3 font-heading text-base font-semibold sm:text-lg">{row.q.text}</p>
  1273→      <p className="mt-1 text-xs text-muted-foreground">
  1274→        You missed this in &ldquo;{row.chapterTitle}&rdquo;
  1275→        {tagLabel ? ` — tagged as ${tagLabel.toLowerCase()}` : ''}.
  1276→      </p>
  1277→
  1278→      <div className="mt-4 rounded-2xl border border-border bg-background p-4">
  1279→        <p className="text-sm font-semibold">Worked solution</p>
  1280→        <ol className="mt-3 flex flex-col gap-3">
  1281→          {steps.map((step, i) => (
  1282→            <li key={step.title} className="flex items-start gap-3">
  1283→              <span
  1284→                className="flex size-7 shrink-0 items-center justify-center rounded-full bg-primary/10 text-xs font-bold text-primary"
  1285→                aria-hidden="true"
  1286→              >
  1287→                {i + 1}
  1288→              </span>
  1289→              <div>
  1290→                <p className="text-sm font-semibold">{step.title}</p>
  1291→                <p className="mt-0.5 text-sm text-muted-foreground">{step.body}</p>
  1292→              </div>
  1293→            </li>
  1294→          ))}
  1295→        </ol>
  1296→      </div>
  1297→
  1298→      <div className="mt-5 flex items-center justify-end">
  1299→        <Button className="min-h-[44px]" onClick={advance}>
  1300→          {last ? 'Finish review' : 'Next mistake'} <ArrowRight className="size-4" aria-hidden="true" />
  1301→        </Button>
  1302→      </div>
  1303→    </TabCard>
  1304→  );
  1305→}
