   152→export function bandOf(cls: string, cat: LearnerCat): Band {
   153→  if (cat === 'dropper') return 'dropper';
   154→  if (cat === 'above') return 'above';
   155→  const n = CLASSES.indexOf(cls);
   156→  if (n < 0) return 'middle';
   157→  if (n <= 2) return 'young';
   158→  const grade = n - 2;
   159→  if (grade <= 5) return 'young';
   160→  if (grade <= 8) return 'middle';
   161→  if (grade <= 10) return 'sec';
   162→  return 'senior';
   163→}

   165→function toMin(hhmm: string, fallback: number): number {
   166→  const m = /^(\d{1,2}):(\d{2})$/.exec(hhmm);
   167→  if (!m) return fallback;
   168→  const h = Number(m[1]);
   169→  const min = Number(m[2]);
   170→  if (h < 0 || h > 23 || min < 0 || min > 59) return fallback;
   171→  return h * 60 + min;
   172→}
   173→
   174→function hoursToMin(hhmm: string, fallback: number): number {
   175→  return toMin(hhmm, fallback);
   176→}

   290→function validateStep(d: WizardDraft, stepId: string): string {
   291→  switch (stepId) {
   292→    case 'class':
   293→      if (d.cat === 'school' && !d.className) return 'Choose your class.';
   294→      return '';
   295→    case 'board':
   296→      if (d.cat !== 'school') return '';
   297→      if (!d.board) return 'Choose your board.';
   298→      if (d.board === 'Other State Board' && !d.stateName.trim()) {
   299→        return 'Tell us your state or union territory so the subject list matches.';
   300→      }
   301→      if (d.board === 'Other / International' && !d.intl.trim()) {
   302→        return 'Describe your curriculum briefly (e.g. IB MYP, Cambridge).';
   303→      }
   304→      return '';
   305→    case 'subjects':
   306→      if (d.cat === 'above') return '';
   307→      if (d.subjects.length === 0 && !d.customSubjects.trim()) {
   308→        return 'Pick at least one subject (or add your own).';
   309→      }
   310→      return '';
   311→    case 'day': {
   312→      const wake = toMin(d.wake, -1);
   313→      const start = toMin(d.schoolStart, -1);
   314→      const end = toMin(d.schoolEnd, -1);
   315→      const dinner = toMin(d.dinner, -1);
   316→      const bed = toMin(d.bedtime, -1);
   317→      if (wake < 0 || start < 0 || end < 0 || dinner < 0 || bed < 0) {
   318→        return 'Enter all times as HH:MM.';
   319→      }
   320→      if (wake > start) return 'Wake time must be before school starts.';
   321→      if (end <= start) return 'School end must be after school start.';
   322→      if (dinner <= end) return 'Dinner must be after school ends.';
   323→      if (bed <= dinner) return 'Bedtime must be after dinner.';
   324→      return '';
   325→    }
   326→    case 'commitments': {
   327→      for (const c of d.commitments) {
   328→        if (c.endMin <= c.startMin) return `“${c.label || 'Commitment'}” must end after it starts.`;
   329→        if (c.endMin - c.startMin > 12 * 60) return 'A commitment cannot be longer than 12 hours.';
   330→      }
   331→      return '';
   332→    }
   333→    case 'outside': {
   334→      if (!d.studyOutside.trim()) return '';
   335→      const n = Number(d.studyOutside);
   336→      if (!Number.isFinite(n) || n < 0 || n > 12) return 'Study hours outside school must be between 0 and 12.';
   337→      return '';
   338→    }
   339→    case 'exams': {
   340→      if (d.nextExamDate) {
   341→        if (Number.isNaN(Date.parse(d.nextExamDate))) return 'Enter a valid exam date.';
   342→        if (Date.parse(d.nextExamDate) < Date.parse(todayStr())) return 'The exam date must be in the future.';
   343→      }
   344→      if (d.compExamDate) {
   345→        if (Number.isNaN(Date.parse(d.compExamDate))) return 'Enter a valid competitive-exam date.';
   346→        if (Date.parse(d.compExamDate) < Date.parse(todayStr())) return 'The competitive-exam date must be in the future.';
   347→      }
   348→      return '';
   349→    }
   350→    case 'dropper-exam': {
   351→      if (!d.exam) return 'Choose the exam you are preparing for.';
   352→      if (d.examDate) {
   353→        if (Number.isNaN(Date.parse(d.examDate))) return 'Enter a valid exam date.';
   354→        if (Date.parse(d.examDate) < Date.parse(todayStr())) return 'The exam date must be in the future.';
   355→      }
   356→      return '';
   357→    }
   358→    case 'dropper-attempts':
   359→      if (!d.attempts) return 'Choose how many attempts you have had.';
   360→      return '';
   361→    case 'dropper-prep': {
   362→      const taken = d.mocksTaken.trim() === '' ? 0 : Number(d.mocksTaken);
   363→      if (!Number.isFinite(taken) || taken < 0 || taken > 50) return 'Mocks taken must be between 0 and 50.';
   364→      if (taken > 0) {
   365→        const score = Number(d.mockScore);
   366→        const total = Number(d.mockTotal);
   367→        if (!Number.isFinite(score) || score < 0) return 'Enter your latest mock score.';
   368→        if (!Number.isFinite(total) || total <= 0) return 'Enter the total the mock is out of.';
   369→        if (score > total) return 'The mock score cannot be above the total.';
   370→      }
   371→      return '';
   372→    }
   373→    case 'dropper-hours': {
   374→      const n = Number(d.availableHours);
   375→      if (!Number.isFinite(n) || n < 1 || n > 16) return 'Available study hours per day must be between 1 and 16.';
   376→      return '';
   377→    }
   378→    case 'above-goal': {
   379→      const focus = d.interest === 'Custom' ? d.interestCustom.trim() : d.interest;
   380→      if (!focus) return 'Choose an area of interest (or write your own goal).';
   381→      if (!d.goalText.trim()) return 'Write your goal in one line.';
   382→      if (d.targetDate) {
   383→        if (Number.isNaN(Date.parse(d.targetDate))) return 'Enter a valid target date.';
   384→        if (Date.parse(d.targetDate) < Date.parse(todayStr())) return 'The target date must be in the future.';
   385→      }
   386→      return '';
   387→    }
   388→    case 'above-time': {
   389→      const n = Number(d.hoursPerWeek);
   390→      if (!Number.isFinite(n) || n < 1 || n > 100) return 'Hours per week must be between 1 and 100.';
   391→      return '';
   392→    }
   393→    case 'guardian':
   394→      return '';
   395→    default:
   396→      return '';
   397→  }
   398→}

   424→    compSub: d.cat === 'school' && d.compSub.trim() ? d.compSub.trim() : undefined,
   425→    guardianConfirmed: d.cat === 'school' ? d.guardianOk : undefined,
   426→    attempts: d.cat === 'dropper' ? d.attempts : undefined,
   427→    syllabusCovered: d.cat === 'dropper' && d.syllabus.trim() ? d.syllabus.trim() : undefined,
   428→    strongAreas: d.cat === 'dropper' && d.strong.trim() ? d.strong.trim() : undefined,
   429→    weakAreas: d.cat === 'dropper' && d.weak.trim() ? d.weak.trim() : undefined,
   430→    prevPrep: d.cat === 'dropper' && d.prevPrep.trim() ? d.prevPrep.trim() : undefined,
   431→    mocksTaken: d.cat === 'dropper' && d.mocksTaken.trim() !== '' ? Number(d.mocksTaken) : undefined,
   432→    mockScore: d.cat === 'dropper' && d.mockScore.trim() !== '' ? Number(d.mockScore) : undefined,
   433→    mockTotal: d.cat === 'dropper' && d.mockTotal.trim() !== '' ? Number(d.mockTotal) : undefined,
   434→    learnerType: d.cat === 'above' ? d.learnerType : undefined,
   435→    interest: d.cat === 'above' ? (d.interest === 'Custom' ? d.interestCustom.trim() : d.interest) : undefined,
   436→    level: d.cat === 'above' ? d.level : undefined,
   437→    targetDate: d.cat === 'above' && d.targetDate ? d.targetDate : undefined,
   438→    hoursPerWeek: d.cat === 'above' && d.hoursPerWeek.trim() !== '' ? Number(d.hoursPerWeek) : undefined,
   439→    wakeMin: hoursToMin(d.wake, 390),
   440→    studyOutsideMin: d.cat === 'school' && d.studyOutside.trim() !== '' ? Math.round(Number(d.studyOutside) * 60) : undefined,
   441→    commitments: d.commitments,
   442→  };
   443→  const plan: SchedulePlan = {
   444→    startMin: hoursToMin(d.schoolStart, 480),
   445→    endMin: hoursToMin(d.schoolEnd, 900),
   446→    dinnerMin: hoursToMin(d.dinner, 1200),
   447→    bedMin: hoursToMin(d.bedtime, 1350),
   448→    sessions: Math.max(1, Math.min(6, BANDS[band].dailyGoal || 2)),
   449→    sessionLen: band === 'young' ? 15 : 30,
   450→    wakeMin: profile.wakeMin,
   451→    busy: d.commitments,
   452→  };
   453→  return { profile, plan };
   454→}

   458→export function Onboarding({
   459→  draft,
   460→  onDraftChange,
   461→  onFinish,
   462→  onCancel,
   463→}: {
   464→  draft: WizardDraft | null;
   465→  onDraftChange: (d: WizardDraft | null) => void;
   466→  onFinish: (r: FirstRunResult) => void;
   467→  /** present when reopening from Settings: keeps the current profile untouched */
   468→  onCancel?: () => void;
   469→}) {
   470→  const [d, setD] = useState<WizardDraft>(draft ?? EMPTY_DRAFT);
   471→  const band = bandOf(d.className, d.cat);
   472→  const steps = stepsFor(d.cat, d.cat === 'school' ? d.className : 'Class 9');
   473→  const stepDef = steps[Math.min(d.step, steps.length - 1)];
   474→  const total = steps.length;
   475→
   476→  const set = (patch: Partial<WizardDraft>) => {
   477→    const next = { ...d, ...patch };
   478→    setD(next);
   479→    onDraftChange(next); // resumable per step
   480→  };
   481→
   482→  const stepError = useMemo(() => validateStep(d, stepDef.id), [d, stepDef.id]);
   483→
   484→  // live schedule preview (7.1 upgrade)
   485→  const preview = useMemo(() => {
   486→    if (stepDef.id !== 'preview') return null;
   487→    const { plan } = answersToResult(d, band);
   488→    const { slots, dropped } = planDay(plan);
   489→    return { plan, slots, dropped };
   490→  }, [stepDef.id, d, band]);

  1137→function CommitmentAdder({ onAdd }: { onAdd: (c: Commitment) => void }) {
  1138→  const [label, setLabel] = useState('');
  1139→  const [start, setStart] = useState('17:00');
  1140→  const [end, setEnd] = useState('18:00');
  1141→  const add = () => {
  1142→    const s = toMin(start, -1);
  1143→    const e = toMin(end, -1);
  1144→    if (!label.trim() || s < 0 || e <= s) return;
  1145→    onAdd({ label: label.trim(), startMin: s, endMin: e });
  1146→    setLabel('');
  1147→    setStart('17:00');
  1148→    setEnd('18:00');
  1149→  };
  1150→  return (
  1151→    <div className="mt-3 grid items-end gap-2 rounded-xl border border-dashed border-border p-3 sm:grid-cols-[2fr_1fr_1fr_auto]">
  1152→      <div>
  1153→        <Label htmlFor="ob-clabel">What</Label>
  1154→        <Input id="ob-clabel" value={label} onChange={(e) => setLabel(e.target.value)} placeholder="e.g. Cricket practice" className="mt-1" />
  1155→      </div>
  1156→      <div>
  1157→        <Label htmlFor="ob-cstart">From</Label>
  1158→        <Input id="ob-cstart" type="time" value={start} onChange={(e) => setStart(e.target.value)} className="mt-1" />
  1159→      </div>
  1160→      <div>
  1161→        <Label htmlFor="ob-cend">To</Label>
